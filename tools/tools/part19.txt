// Lambda calibration v9.6 — two-phase grid search (§3.3) +
// hybrid prior blending (v9) + NEW multi-market WLS blend (§8).
// Multi-market adds DC / O1.5 / O2.5 / BTTS implied probs to the
// error metric so λ reflects the book's GOALS prices too —
// fixing the systematic BTTS-Yes bias seen in the 11/08 review.
// Standard engine only; jackpot stays 1X2-pure (Rule 6).
// ============================================================
import { buildScoreMatrix, outcomeProbs } from './dixon-coles';
import type { CalibrationResult, HybridCalibration, TeamStrengthPrior } from '../atlas-types';

export function stripMargin(h: number, d: number, a: number): { probs: [number, number, number]; margin: number } {
  const ih = 1 / h, id = 1 / d, ia = 1 / a;
  const s = ih + id + ia;
  return { probs: [ih / s, id / s, ia / s], margin: s - 1 };
}

export interface ExtraMarketProbs {
  p1X?: number; pX2?: number; pO15?: number; pO25?: number; pBTTS?: number;
  count: number;
}

function modelAt(lH: number, lA: number, rho: number) {
  const sm = buildScoreMatrix(lH, lA, rho);
  const p = outcomeProbs(sm.raw, sm.sum);
  let le1 = 0, le2 = 0, btts = 0;
  for (const e of sm.matrix) {
    const t = e.i + e.j;
    if (t <= 1) le1 += e.p;
    if (t <= 2) le2 += e.p;
    if (e.i > 0 && e.j > 0) btts += e.p;
  }
  return { pH: p.home, pD: p.draw, pA: p.away, o15: 1 - le1, o25: 1 - le2, btts };
}

function targetWithPrior(probs: [number, number, number], margin: number, prior?: TeamStrengthPrior) {
  let target = probs;
  let hybrid: HybridCalibration = {
    source: 'market-only', marketProbs: probs, priorProbs: null,
    blendedProbs: probs, marketPrecision: 1 / Math.max(margin, 0.01), priorPrecision: 0,
  };
  if (prior && prior.confidence > 0) {
    const wm = 1 / Math.max(margin, 0.01);
    const wp = Math.min(prior.confidence, 8);
    const pp: [number, number, number] = [prior.pHome, prior.pDraw, prior.pAway];
    const b = probs.map((p, i) => (p * wm + pp[i] * wp) / (wm + wp)) as [number, number, number];
    const s = b[0] + b[1] + b[2];
    target = [b[0] / s, b[1] / s, b[2] / s];
    hybrid = { source: 'hybrid', marketProbs: probs, priorProbs: pp, blendedProbs: target, marketPrecision: wm, priorPrecision: wp };
  }
  return { target, hybrid };
}

function gridSearch(err: (lH: number, lA: number) => number) {
  let bH = 1.2, bA = 1.0, bE = Infinity;
  for (let lH = 0.3; lH <= 3.01; lH += 0.1)
    for (let lA = 0.2; lA <= 2.51; lA += 0.1) {
      const e = err(lH, lA);
      if (e < bE) { bE = e; bH = lH; bA = lA; }
    }
  for (let lH = Math.max(0.2, bH - 0.1); lH <= bH + 0.11; lH += 0.01)
    for (let lA = Math.max(0.1, bA - 0.1); lA <= bA + 0.11; lA += 0.01) {
      const e = err(lH, lA);
      if (e < bE) { bE = e; bH = lH; bA = lA; }
    }
  return { lambdaH: Math.round(bH * 100) / 100, lambdaA: Math.round(bA * 100) / 100, sse: bE };
}

export function calibrateLambda(h: number, d: number, a: number, rho: number, prior?: TeamStrengthPrior): CalibrationResult {
  const { probs, margin } = stripMargin(h, d, a);
  const { target, hybrid } = targetWithPrior(probs, margin, prior);
  const res = gridSearch((lH, lA) => {
    const m = modelAt(lH, lA, rho);
    return (m.pH - target[0]) ** 2 + (m.pD - target[1]) ** 2 + (m.pA - target[2]) ** 2;
  });
  return { ...res, hybrid };
}

// v9.6 §8 — weighted least squares across all available markets
export function calibrateLambdaMultiMarket(
  h: number, d: number, a: number, rho: number,
  extra: ExtraMarketProbs, prior?: TeamStrengthPrior,
): CalibrationResult {
  const { probs, margin } = stripMargin(h, d, a);
  const { target, hybrid } = targetWithPrior(probs, margin, prior);
  const wDC = 0.5, wG = 0.9, wB = 0.9;
  const res = gridSearch((lH, lA) => {
    const m = modelAt(lH, lA, rho);
    let e = (m.pH - target[0]) ** 2 + (m.pD - target[1]) ** 2 + (m.pA - target[2]) ** 2;
    if (extra.p1X !== undefined) e += wDC * (m.pH + m.pD - extra.p1X) ** 2;
    if (extra.pX2 !== undefined) e += wDC * (m.pD + m.pA - extra.pX2) ** 2;
    if (extra.pO15 !== undefined) e += wG * (m.o15 - extra.pO15) ** 2;
    if (extra.pO25 !== undefined) e += wG * (m.o25 - extra.pO25) ** 2;
    if (extra.pBTTS !== undefined) e += wB * (m.btts - extra.pBTTS) ** 2;
    return e;
  });
  return { ...res, hybrid, mmMarkets: extra.count };
}

// ============================================================
// STANDARD ENGINE v9.6 — EV-maximizing, rho = -0.13 (RULE 1)
// v9.6: when ≥3 extended prices exist (PDF/Smart-Paste), λ is
// calibrated via multi-market WLS (§8) — kills the systematic
// BTTS-Yes bias; otherwise falls back to 1X2-only calibration.
// Honest confidence, BET/NO-BET verdict, tightGame flag (v9.1).
// ============================================================
import { DC_RHO_STANDARD } from '../atlas-types';
import type {
  Match, Outcome, AtlasAnalysis, MarginWarning, PortfolioOutcome, DecorrelationResult, Verdict,
} from '../atlas-types';
import { calibrateLambda, calibrateLambdaMultiMarket, stripMargin } from '../math/calibration';
import type { ExtraMarketProbs } from '../math/calibration';
import { buildScoreMatrix, outcomeProbs } from '../math/dixon-coles';
import { adaptiveSimulate } from '../math/adaptive-mc';
import { htftCopulaJoint } from '../math/copula';
import { applyLearnedAdjustments, defaultParams } from '../learning/bayesian-learning';
import type { LearnedParams } from '../learning/bayesian-learning';

function marginWarning(minOdds: number): MarginWarning {
  if (minOdds < 1.25) return { level: 'severe', msg: 'Extreme favorite — bookmaker margin asymmetry severe. Avoid.' };
  if (minOdds < 1.45) return { level: 'high', msg: 'Heavy favorite — elevated bookmaker edge on short side.' };
  if (minOdds < 1.65) return { level: 'moderate', msg: 'Moderate favorite — slightly elevated margin.' };
  return { level: 'none', msg: '' };
}

function extraTargets(u: Record<string, number>): ExtraMarketProbs {
  const imp = (o: number) => (o > 1 ? 1 / o : 0);
  const x: ExtraMarketProbs = { count: 0 };
  const i1X = imp(u.dc1X), iX2 = imp(u.dcX2), i12 = imp(u.dc12);
  const sDC = i1X + iX2 + i12;
  if (sDC > 0 && i1X && iX2) { x.p1X = (2 * i1X) / sDC; x.pX2 = (2 * iX2) / sDC; x.count += 2; }
  const iO = imp(u.over25), iU = imp(u.under25);
  if (iO && iU) { x.pO25 = iO / (iO + iU); x.count++; }
  const iO15 = imp(u.over15), iU15 = imp(u.under15);
  if (iO15 && iU15) { x.pO15 = iO15 / (iO15 + iU15); x.count++; }
  const iY = imp(u.bttsYes), iN = imp(u.bttsNo);
  if (iY && iN) { x.pBTTS = iY / (iY + iN); x.count++; }
  return x;
}

export function atlasAnalyseV9(
  match: Match, phi = 0.10, learnedParams: LearnedParams = defaultParams(),
): AtlasAnalysis {
  const h = parseFloat(match.h), d = parseFloat(match.d), a = parseFloat(match.a);
  if (!(h > 1 && d > 1 && a > 1)) throw new Error('Invalid 1X2 odds');

  const { probs: bookProbs, margin } = stripMargin(h, d, a);
  const extra = extraTargets(match.userOdds);
  const cal = extra.count >= 3
    ? calibrateLambdaMultiMarket(h, d, a, DC_RHO_STANDARD, extra, match.prior)
    : calibrateLambda(h, d, a, DC_RHO_STANDARD, match.prior);
  const rho = DC_RHO_STANDARD + learnedParams.rhoAdjust;

  const sm = buildScoreMatrix(cal.lambdaH, cal.lambdaA, rho);
  const dc = outcomeProbs(sm.raw, sm.sum);

  const tightGame = Math.abs(dc.home - dc.away) < 0.08;

  let pick: Outcome = (Object.entries(dc) as [Outcome, number][]).sort((x, y) => y[1] - x[1])[0][0];
  let confidence = dc[pick];

  const sofT = learnedParams.sofThreshold;
  const sofPrimary = Math.abs(dc.home - dc.away) < sofT;
  const sofSecondary = Math.abs(dc.home - dc.draw) < 0.03 || Math.abs(dc.draw - dc.away) < 0.03;
  const drawShortest = d < h && d < a;
  const sofTriggered = sofPrimary || sofSecondary || drawShortest;
  if (sofTriggered) {
    pick = 'draw';
    confidence = dc.draw;
  }

  const edges: Record<Outcome, number> = {
    home: dc.home - bookProbs[0], draw: dc.draw - bookProbs[1], away: dc.away - bookProbs[2],
  };
  const oddsMap: Record<Outcome, number> = { home: h, draw: d, away: a };
  const evs: Record<Outcome, number> = {
    home: dc.home * h - 1, draw: dc.draw * d - 1, away: dc.away * a - 1,
  };
  const spots = (Object.keys(edges) as Outcome[]).filter((o) => edges[o] > 0 && evs[o] > 0);
  const spotted = spots.sort((x, y) => edges[y] - edges[x])[0] ?? null;
  const maxEdge = Math.max(...Object.values(edges), 0);
  const decorrelation: DecorrelationResult = {
    score: Math.min(100, maxEdge * 500), edges, evs, spotted,
    hasRealSpot: !!spotted, maxEdge,
  };
  let decorrOverride = false;
  if (spotted && spotted !== pick && decorrelation.score >= 15 && edges[spotted] > 0.04 && evs[spotted] > 0.03) {
    pick = spotted; decorrOverride = true;
    confidence = dc[pick];
  }
  confidence = Math.min(confidence, 0.78);

  const sim = adaptiveSimulate(cal.lambdaH, cal.lambdaA, DC_RHO_STANDARD, {
    gridProbs: dc, gridPick: pick,
  });

  const [fh, fd, fa] = applyLearnedAdjustments(
    dc.home, dc.draw, dc.away, match.league, learnedParams, sim.confMultiplier,
  );
  const probs: Record<Outcome, number> = { home: fh, draw: fd, away: fa };

  const portfolio: PortfolioOutcome[] = (['home', 'draw', 'away'] as Outcome[]).map((o) => {
    const ev = probs[o] * oddsMap[o] - 1;
    const variance = (1 - probs[o]) * probs[o] * oddsMap[o] ** 2;
    const kelly = ev > 0 ? Math.min(ev / (oddsMap[o] - 1), 0.25) : 0;
    return { outcome: o, stake: kelly, ev, sharpe: variance > 0 ? ev / Math.sqrt(variance) : 0,
             kelly, isMaxEp: false, varPerUnit: variance };
  });
  const best = portfolio.filter((p) => p.ev > 0).sort((x, y) => y.ev - x.ev)[0];
  if (best) best.isMaxEp = true;

  const htft = htftCopulaJoint(cal.lambdaH, cal.lambdaA, DC_RHO_STANDARD);
  const thresholdPass = Math.abs(probs[pick] - 0.5) >= phi;

  const bestEv = Math.max(...portfolio.map((p) => p.ev));
  const verdict: Verdict = thresholdPass && bestEv > 0 ? 'BET' : 'NO-BET';

  return {
    pick, confidence, probs, dcProbs: dc,
    bookProbs: { home: bookProbs[0], draw: bookProbs[1], away: bookProbs[2] },
    margin, marginWarning: marginWarning(Math.min(h, d, a)),
    calibration: cal, scoreMatrix: sm.matrix, decorrelation, portfolio,
    sofTriggered, decorrOverride, thresholdPass, verdict, tightGame,
    sim, htft, engine: 'standard',
  };
}
