// ============================================================
// Smart Paste v4 — FULLY SECTION-SCOPED (SportPesa market page).
// v9.8 fixes the label-collision leaks: "OVER 2.50" under
// AWAY TEAM maps to awayOver25, never over25; "YES (GG)" under
// FIRST HALF maps to bttsHTYes, never bttsYes. The 3 WAY section
// refreshes the main H/D/A odds. Free-form labeled pastes
// ("GG: 1.80") still work via the legacy map when no header seen.
// ============================================================
export type PasteProfile = 'auto' | 'sportpesa' | 'betika';
export interface ParseOpts { profile?: PasteProfile; home?: string; away?: string; }
export interface ParsedOdds {
  odds: Record<string, number>; count: number; unmatched: string[];
  main?: { h: number; d: number; a: number };
}

const norm = (s: string) => s.toLowerCase().replace(/\s+/g, ' ').trim();
const stripB = (s: string) => s.replace(/[\[\]]/g, '');

const SECTION_RE: { re: RegExp; id: string }[] = [
  { re: /1x2\s*\(\s*\d?\s*up\s*\)/, id: 'skip' },
  { re: /3 way\s*-\s*first half/, id: 'ht3' },
  { re: /3 way/, id: 'main' },
  { re: /both teams to score\s*-\s*first half/, id: 'bttsHT' },
  { re: /both teams to score/, id: 'btts' },
  { re: /double chance/, id: 'dc' },
  { re: /draw no bet/, id: 'skip' },
  { re: /asian total goals over\/under\s*-\s*half time/, id: 'athtHT' },
  { re: /asian total goals/, id: 'skip' },
  { re: /over\/under home team/, id: 'homeOU' },
  { re: /over\/under away team/, id: 'awayOU' },
  { re: /over\/under\s*-\s*half time/, id: 'ouHT' },
  { re: /over\/under\s*-\s*full time/, id: 'ouFT' },
  { re: /correct score\s*-\s*first half/, id: 'htcs' },
  { re: /correct score/, id: 'cs' },
  { re: /half with most goals/, id: 'hmost' },
  { re: /half time\/full time/, id: 'htft' },
  { re: /euro handicap/, id: 'eh' },
  { re: /result\s*\+\s*total goals/, id: 'ftrOU' },
  { re: /result\s*\+\s*both teams/, id: 'ftrBTTS' },
  { re: /first team to score/, id: 'fts' },
  { re: /odd\s*\/?\s*even/, id: 'oe' },
];

const HTFT_ALIASES: Record<string, string> = {
  '1/1': 'htft_1_1', '1/x': 'htft_1_X', '1/2': 'htft_1_2',
  'x/1': 'htft_X_1', 'x/x': 'htft_X_X', 'x/2': 'htft_X_2',
  '2/1': 'htft_2_1', '2/x': 'htft_2_X', '2/2': 'htft_2_2',
};

function ouKey(dir: 'over' | 'under', line: number, prefix: '' | 'ht' | 'home' | 'away' | 'atht'): string | null {
  const lk = String(line).replace('.', '');
  const allowed: Record<string, string[]> = {
    '': ['05', '15', '25', '35'],
    ht: ['05', '15'],
    home: ['05', '15', '25'],
    away: ['05', '15', '25'],
    atht: ['075', '100'],
  };
  const map: Record<string, string> = { '0.5': '05', '1.5': '15', '2.5': '25', '3.5': '35', '0.75': '075', '1': '100', '1.0': '100', '1.00': '100' };
  const k = map[String(line)];
  if (!k || !allowed[prefix].includes(k)) return null;
  const base = prefix === '' ? '' : prefix;
  return `${base}${dir === 'over' ? 'Over' : 'Under'}${k}`;
}

function resolveIn(sec: string, label: string, rowInSec: number): string | null {
  const l = norm(stripB(label));
  const ou = l.match(/^(over|under)\s*(\d+(?:\.\d+)?)$/);
  const isD = /^(draw|x)$/.test(l);
  switch (sec) {
    case 'btts': return /yes|gg/.test(l) ? 'bttsYes' : /no|ng/.test(l) ? 'bttsNo' : null;
    case 'bttsHT': return /yes|gg/.test(l) ? 'bttsHTYes' : /no|ng/.test(l) ? 'bttsHTNo' : null;
    case 'dc':
      if (/1 or x|^1x$/.test(l)) return 'dc1X';
      if (/x or 2|^x2$/.test(l)) return 'dcX2';
      if (/1 or 2|^12$/.test(l)) return 'dc12';
      return null;
    case 'ouFT': return ou ? ouKey(ou[1] as 'over' | 'under', parseFloat(ou[2]), '') : null;
    case 'ouHT': return ou ? ouKey(ou[1] as 'over' | 'under', parseFloat(ou[2]), 'ht') : null;
    case 'homeOU': return ou ? ouKey(ou[1] as 'over' | 'under', parseFloat(ou[2]), 'home') : null;
    case 'awayOU': return ou ? ouKey(ou[1] as 'over' | 'under', parseFloat(ou[2]), 'away') : null;
    case 'athtHT': return ou ? ouKey(ou[1] as 'over' | 'under', parseFloat(ou[2]), 'atht') : null;
    case 'fts':
      if (/neither/.test(l)) return 'ftsNeither';
      if (/^1$|home/.test(l)) return 'ftsHome';
      if (/^2$|away/.test(l)) return 'ftsAway';
      return null;
    case 'ht3':
      if (isD) return 'ht3Draw';
      return rowInSec === 0 ? 'ht3Home' : 'ht3Away';
    case 'hmost':
      if (/^1st/.test(l)) return 'halfMost1st';
      if (/^2nd/.test(l)) return 'halfMost2nd';
      if (/neither|equal/.test(l)) return 'halfMostNeither';
      return null;
    case 'htft': return HTFT_ALIASES[l] ?? null;
    case 'eh': {
      const m = l.match(/^(1|x|2)\s*\(\s*(-?\d+)\s*:\s*(-?\d+)\s*\)$/);
      if (!m) return null;
      const out = m[1] === 'x' ? 'X' : m[1] === '1' ? '1' : '2';
      return `eh_${m[2]}:${m[3]}_${out}`;
    }
    case 'cs':
    case 'htcs': {
      const pre = sec === 'cs' ? 'cs' : 'htcs';
      if (/any other/.test(l)) return `${pre}_any`;
      const m = l.match(/^(\d+)\s*[:\-]\s*(\d+)$/);
      if (m && +m[1] <= 9 && +m[2] <= 9) return `${pre}_${m[1]}_${m[2]}`;
      return null;
    }
    case 'ftrOU': {
      const m = l.match(/^(1|x|2)\s*(?:&|and)\s*(over|under)\s*(\d\.5)$/);
      if (!m) return null;
      const f = m[1] === 'x' ? 'X' : m[1];
      return `ftrOU_${f}_${m[2] === 'over' ? 'O' : 'U'}${m[3].replace('.', '')}`;
    }
    case 'ftrBTTS': {
      const m = l.match(/^(1|x|2)\s*(?:&|and)\s*(yes|no)/);
      if (!m) return null;
      const f = m[1] === 'x' ? 'X' : m[1];
      return `ftrBTTS_${f}_${m[2] === 'yes' ? 'Yes' : 'No'}`;
    }
    case 'oe': return /^odd/.test(l) ? 'oddGoals' : /^even/.test(l) ? 'evenGoals' : null;
    default: return null;
  }
}

// legacy free-form map (used only before any section header is seen)
const LABEL_MAP: Record<string, string> = {
  'gg': 'bttsYes', 'btts': 'bttsYes', 'btts yes': 'bttsYes', 'yes (gg)': 'bttsYes',
  'ng': 'bttsNo', 'btts no': 'bttsNo', 'no (ng)': 'bttsNo',
  'over 0.5': 'over05', 'under 0.5': 'under05', 'over 1.5': 'over15', 'under 1.5': 'under15',
  'over 2.5': 'over25', 'under 2.5': 'under25', 'over 3.5': 'over35', 'under 3.5': 'under35',
  '1x': 'dc1X', 'x2': 'dcX2', '12': 'dc12', '1 or x': 'dc1X', 'x or 2': 'dcX2', '1 or 2': 'dc12',
  'odd': 'oddGoals', 'even': 'evenGoals',
};

export function parseSmartPaste(text: string, _opts: ParseOpts = {}): ParsedOdds {
  const odds: Record<string, number> = {};
  const unmatched: string[] = [];
  let main: { h: number; d: number; a: number } | undefined;
  if (!text || !text.trim()) return { odds, count: 0, unmatched };

  const lines = text.split(/\n/).map((l) => l.trim()).filter(Boolean);
  let sec: string | null = null;
  let rowInSec = 0;

  for (const line of lines) {
    if (line.startsWith('|') || /\d{2}-\d{2}-\d{4}/.test(line)) continue;
    const low = norm(stripB(line));
    const hdr = SECTION_RE.find((s) => s.re.test(low));
    if (hdr && !/\d\.\d+/.test(line)) { sec = hdr.id; rowInSec = 0; continue; }

    const m = line.match(/^(.+?)\s*[:\t]\s*(\d+\.\d+)\s*$/) ||
              line.match(/^(.+?)\s{2,}(\d+\.\d+)\s*$/) ||
              (sec ? line.match(/^(.+?)\s+(\d+\.\d+)$/) : null);
    if (!m) continue;
    const label = m[1];
    const val = parseFloat(m[2]);

    if (sec === 'skip') continue;
    if (sec === 'main') {
      const l = norm(label);
      if (/draw|x/.test(l)) main = { ...(main ?? { h: 0, d: 0, a: 0 }), d: val };
      else if (/away|2/.test(l) || (main && main.h && main.d && !main.a)) main = { ...(main ?? { h: 0, d: 0, a: 0 }), a: val };
      else main = { ...(main ?? { h: 0, d: 0, a: 0 }), h: val };
      continue;
    }
    if (sec) {
      const key = resolveIn(sec, label, rowInSec);
      if (key) odds[key] = val; else unmatched.push(line);
      rowInSec++;
      continue;
    }
    // free-form (no header seen yet)
    const l = norm(stripB(label));
    const key = LABEL_MAP[l] ?? HTFT_ALIASES[l] ?? null;
    if (key) odds[key] = val; else unmatched.push(line);
  }
  if (main && !(main.h > 1 && main.d > 1 && main.a > 1)) main = undefined;
  return { odds, count: Object.keys(odds).length, unmatched, main };
}

'use client';
import { useMemo, useState } from 'react';
import { parseSmartPaste } from '@/lib/v9/parsers/odds-parser';
import type { PasteProfile } from '@/lib/v9/parsers/odds-parser';
import { useAtlasStore } from '@/lib/v9/store/atlas-store';

export default function SmartPasteOdds({ matchId, onClose }: { matchId: string; onClose: () => void }) {
  const [text, setText] = useState('');
  const [profile, setProfile] = useState<PasteProfile>('auto');
  const updateOdds = useAtlasStore((s) => s.updateOdds);
  const updateMatch = useAtlasStore((s) => s.updateMatch);

  const parsed = useMemo(
    () => parseSmartPaste(text, { profile }),
    [text, profile],
  );

  const warnings = useMemo(() => {
    const w: string[] = [];
    const imp = (o?: number) => (o && o > 1 ? 1 / o : 0);
    const o = parsed.odds;
    if (o.bttsYes && o.bttsNo) {
      const s = imp(o.bttsYes) + imp(o.bttsNo);
      if (s < 0.98 || s > 1.3) w.push(`BTTS Yes+No implied ${(s * 100).toFixed(0)}% (sane 100–130%) — check mapping.`);
    }
    if (o.over25 && o.under25) {
      const s = imp(o.over25) + imp(o.under25);
      if (s < 0.98 || s > 1.3) w.push(`Over+Under 2.5 implied ${(s * 100).toFixed(0)}% — check mapping.`);
    }
    return w;
  }, [parsed]);

  const apply = () => {
    Object.entries(parsed.odds).forEach(([k, v]) => updateOdds(matchId, k, v));
    if (parsed.main) {
      updateMatch(matchId, { h: String(parsed.main.h), d: String(parsed.main.d), a: String(parsed.main.a) });
    }
    onClose();
  };

  const pill = (p: PasteProfile, label: string) => (
    <button onClick={() => setProfile(p)}
      className={`rounded-full px-2.5 py-0.5 text-[10px] font-bold border ${
        profile === p ? 'bg-blue-500/20 text-blue-300 border-blue-500/40' : 'bg-zinc-800 text-zinc-500 border-zinc-700'}`}>
      {label}
    </button>
  );

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/70 p-4" onClick={onClose}>
      <div className="w-full max-w-lg rounded-xl border border-zinc-700 bg-zinc-900 p-4" onClick={(e) => e.stopPropagation()}>
        <div className="mb-2 flex items-center justify-between">
          <h3 className="text-sm font-semibold text-zinc-200">Smart Paste Odds (v4 section-scoped)</h3>
          <span className={`rounded-full px-2.5 py-0.5 text-xs font-bold tabular-nums ${
            parsed.count > 0 ? 'bg-emerald-500/15 text-emerald-400' : 'bg-zinc-800 text-zinc-500'}`}>
            {parsed.count} markets detected
          </span>
        </div>

        <div className="mb-2 flex items-center gap-1.5">
          <span className="text-[10px] uppercase text-zinc-500">Source:</span>
          {pill('auto', 'Auto')}
          {pill('sportpesa', 'SportPesa')}
          {pill('betika', 'Betika')}
          {parsed.main && <span className="ml-auto text-[10px] text-emerald-400">3-Way: {parsed.main.h}/{parsed.main.d}/{parsed.main.a}</span>}
        </div>

        <textarea
          value={text} onChange={(e) => setText(e.target.value)}
          placeholder={'Paste the full market page — headers included.\nBOTH TEAMS TO SCORE\nYES (GG) 1.72\nNO (NG) 2.01'}
          className="h-40 w-full rounded-lg border border-zinc-700 bg-zinc-950 p-2 font-mono text-xs text-zinc-300 focus:border-blue-500 focus:outline-none"
        />

        {warnings.length > 0 && (
          <div className="mt-2 space-y-1 rounded-lg border border-amber-500/30 bg-amber-500/10 p-2">
            {warnings.map((w, i) => <div key={i} className="text-[10px] text-amber-400">⚠ {w}</div>)}
          </div>
        )}
        {parsed.unmatched.length > 0 && (
          <div className="mt-1 text-[10px] text-zinc-500">{parsed.unmatched.length} unmatched line(s)</div>
        )}

        <div className="mt-3 flex justify-end gap-2">
          <button onClick={onClose} className="rounded-lg px-3 py-1.5 text-xs text-zinc-400 hover:bg-zinc-800">Cancel</button>
          <button onClick={apply} disabled={!parsed.count && !parsed.main}
            className="rounded-lg bg-blue-600 px-3 py-1.5 text-xs font-semibold text-white disabled:opacity-40 hover:bg-blue-500">
            Parse & Apply {parsed.count} odds
          </button>
        </div>
      </div>
    </div>
  );
}
