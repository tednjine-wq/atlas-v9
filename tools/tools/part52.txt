// ============================================================
// Context Agent v1 (semi-auto) — Gemini + Google Search
// grounding fetches REAL-TIME team context: motivation,
// key absences, form, fatigue, weather, derby flag.
// Human reviews every result; the engine math is NEVER
// mutated by the agent (KB Rule 5 spirit: model stays primary).
// ============================================================
export interface TeamContext {
  motivation: 'high' | 'medium' | 'low';
  reason: string;
  formLast5: string;
  keyAbsences: string[];
  fatigue: 'none' | 'medium' | 'high';
}
export interface MatchContext {
  home: TeamContext; away: TeamContext;
  weather: string; derby: boolean;
  summary: string; confidence: number; fetchedAt: string;
}

const MODEL = 'gemini-2.5-flash';

export async function fetchMatchContext(
  home: string, away: string, league: string, kickoff: string, apiKey: string,
): Promise<MatchContext> {
  const today = new Date().toISOString().slice(0, 10);
  const prompt = `You are a football intelligence analyst. Today is ${today}.
Match: ${home} vs ${away} (${league}), kickoff ${kickoff}.
Use fresh web evidence (last 14 days only). Return STRICT JSON, no prose:
{"home":{"motivation":"high|medium|low","reason":"max 18 words","formLast5":"e.g. W-D-L-W-W","keyAbsences":["player (reason)"],"fatigue":"none|medium|high"},
"away":{"motivation":"high|medium|low","reason":"max 18 words","formLast5":"e.g. W-W-D-L-W","keyAbsences":["player (reason)"],"fatigue":"none|medium|high"},
"weather":"max 8 words","derby":true|false,
"summary":"max 40 words, betting-relevant synthesis","confidence":0.0-1.0}`;
  const res = await fetch(
    `https://generativelanguage.googleapis.com/v1beta/models/${MODEL}:generateContent?key=${encodeURIComponent(apiKey)}`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        contents: [{ parts: [{ text: prompt }] }],
        tools: [{ google_search: {} }],
        generationConfig: { temperature: 0.2, responseMimeType: 'application/json' },
      }),
    },
  );
  if (!res.ok) throw new Error(`Gemini API ${res.status} — check key / quota / model name`);
  const json = await res.json();
  const text: string = json?.candidates?.[0]?.content?.parts?.[0]?.text ?? '';
  const m = text.match(/\{[\s\S]*\}/);
  if (!m) throw new Error('Model returned no JSON');
  return { ...(JSON.parse(m[0]) as MatchContext), fetchedAt: new Date().toLocaleString() };
}

export function contextVerdict(ctx: MatchContext): string {
  const h = ctx.home.motivation, a = ctx.away.motivation;
  const hAbs = ctx.home.keyAbsences.length, aAbs = ctx.away.keyAbsences.length;
  if (h === 'low' && a !== 'low') return 'Context favors AWAY (home motivation low)';
  if (a === 'low' && h !== 'low') return 'Context favors HOME (away motivation low)';
  if (hAbs > aAbs + 1) return 'Context favors AWAY (home key absences)';
  if (aAbs > hAbs + 1) return 'Context favors HOME (away key absences)';
  if (h === 'high' && a === 'high') return 'High-stakes clash — tight game, draw risk elevated';
  if (ctx.derby) return 'Derby — form guides weaken, variance up';
  return 'Context broadly neutral — trust the engine';
}

'use client';
import { useState } from 'react';
import { useAtlasStore } from '@/lib/v9/store/atlas-store';
import { fetchMatchContext, contextVerdict } from '@/lib/v9/agents/context-agent';
import type { MatchContext } from '@/lib/v9/agents/context-agent';

const KEY = 'atlas_context_key';
const motColor = (m: string) =>
  m === 'high' ? 'text-emerald-400' : m === 'low' ? 'text-rose-400' : 'text-amber-400';

function TeamBlock({ label, t }: { label: string; t: MatchContext['home'] }) {
  return (
    <div className="rounded-lg border border-zinc-800 bg-zinc-950/60 p-2">
      <div className="mb-1 flex items-center justify-between">
        <span className="text-[11px] font-bold text-zinc-200">{label}</span>
        <span className={`text-[10px] font-bold uppercase ${motColor(t.motivation)}`}>{t.motivation} motivation</span>
      </div>
      <div className="text-[10px] text-zinc-400">{t.reason}</div>
      <div className="mt-1 flex gap-3 text-[10px] text-zinc-500 tabular-nums">
        <span>form {t.formLast5 || '—'}</span>
        <span>fatigue {t.fatigue}</span>
      </div>
      {t.keyAbsences.length > 0 && (
        <div className="mt-1 text-[10px] text-rose-300">Out: {t.keyAbsences.join(', ')}</div>
      )}
    </div>
  );
}

export default function ContextLabPanel() {
  const matches = useAtlasStore((s) => s.matches);
  const [apiKey, setApiKey] = useState(() =>
    typeof window !== 'undefined' ? localStorage.getItem(KEY) ?? '' : '');
  const [matchId, setMatchId] = useState('');
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState('');
  const [ctx, setCtx] = useState<MatchContext | null>(null);
  const [verdict, setVerdict] = useState('');
  const [history, setHistory] = useState<{ label: string; verdict: string }[]>([]);

  const target = matches.find((m) => m.id === matchId) ?? matches[0];

  const run = async () => {
    if (!target) { setErr('Add a match first (Matches tab).'); return; }
    if (!apiKey.trim()) { setErr('Paste your Google AI Studio API key first.'); return; }
    setBusy(true); setErr(''); setCtx(null);
    try {
      const c = await fetchMatchContext(target.home, target.away, target.league, target.kickoff, apiKey.trim());
      const v = contextVerdict(c);
      setCtx(c); setVerdict(v);
      setHistory((h) => [{ label: `${target.home} v ${target.away}`, verdict: v }, ...h].slice(0, 12));
    } catch (e) {
      setErr(e instanceof Error ? e.message : 'Fetch failed');
    } finally { setBusy(false); }
  };

  return (
    <div className="space-y-3">
      <div className="rounded-xl border border-zinc-800 bg-zinc-900/60 p-3">
        <h3 className="mb-2 text-xs font-bold text-zinc-300">CONTEXT LAB (semi-auto · Gemini + Google Search · you decide)</h3>
        <div className="flex flex-wrap gap-2">
          <input
            type="password" value={apiKey}
            onChange={(e) => { setApiKey(e.target.value); localStorage.setItem(KEY, e.target.value); }}
            placeholder="Google AI Studio API key (aistudio.google.com/apikey)"
            className="w-72 rounded-lg border border-zinc-700 bg-zinc-950 px-2 py-1.5 text-xs text-zinc-300 focus:border-teal-500 focus:outline-none"
          />
          <select value={target?.id ?? ''} onChange={(e) => setMatchId(e.target.value)}
            className="max-w-64 rounded-lg border border-zinc-700 bg-zinc-950 px-2 py-1.5 text-xs text-zinc-300">
            {matches.map((m) => <option key={m.id} value={m.id}>{m.home} v {m.away}</option>)}
          </select>
          <button onClick={run} disabled={busy}
            className="rounded-lg bg-teal-600 px-3 py-1.5 text-xs font-semibold text-white hover:bg-teal-500 disabled:opacity-40">
            {busy ? 'Fetching live context…' : '🌐 Fetch live context'}
          </button>
        </div>
        {err && <div className="mt-2 text-[11px] text-rose-400">{err}</div>}
      </div>

      {ctx && target && (
        <div className="rounded-xl border border-teal-500/25 bg-zinc-900/60 p-3">
          <div className="mb-2 text-xs font-bold text-teal-300">
            {target.home} v {target.away} · <span className="uppercase">{verdict}</span>
          </div>
          <div className="grid gap-2 md:grid-cols-2">
            <TeamBlock label={target.home} t={ctx.home} />
            <TeamBlock label={target.away} t={ctx.away} />
          </div>
          <div className="mt-2 text-[11px] text-zinc-300">{ctx.summary}</div>
          <div className="mt-1 flex flex-wrap gap-3 text-[10px] text-zinc-500">
            <span>weather: {ctx.weather || '—'}</span>
            <span>derby: {ctx.derby ? 'yes' : 'no'}</span>
            <span>source confidence: {(ctx.confidence * 100).toFixed(0)}%</span>
            <span>fetched {ctx.fetchedAt}</span>
          </div>
          <div className="mt-2 text-[10px] text-amber-400">
            Advisory only — compare with the engine pick on the match card, then YOU decide (star / skip / override).
          </div>
        </div>
      )}

      {history.length > 0 && (
        <div className="rounded-xl border border-zinc-800 bg-zinc-900/60 p-3">
          <h3 className="mb-2 text-xs font-bold text-zinc-300">CONTEXT HISTORY ({history.length})</h3>
          <div className="space-y-1">
            {history.map((h, i) => (
              <div key={i} className="flex items-center justify-between rounded bg-zinc-950/60 px-2 py-1 text-[11px]">
                <span className="text-zinc-300">{h.label}</span>
                <span className="text-teal-400">{h.verdict}</span>
              </div>
            ))}
          </div>
        </div>
      )}
    </div>
  );
}
