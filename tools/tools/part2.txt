// Extended Markets Engine v9.0 — 23 market groups, 70+ odds keys
// V8.7 AUDIT FIXES PRESERVED (Knowledge Base 10.1):
//  • BTTS sums the DC-corrected matrix (NOT independence approx)
//  • AH quarter-lines use the integer-line push rule (not naive avg)
//  • AH sign convention: home covers when margin > -line
// V9.0 UPGRADE: HT/FT uses Gaussian-copula joint refined by IPF
// to exact analytic HT/FT margins (replaces independent halves)
// RULE: score matrix computed ONCE upstream, passed in here
// ============================================================
import { buildScoreMatrix, outcomeProbs } from '../math/dixon-coles';
import { poisson } from '../math/poisson';
import { htftCopulaJoint } from '../math/copula';
import type { ScoreEntry } from '../atlas-types';

export interface ExtendedMarketItem {
  key: string; label: string; prob: number;
  odds: number; ev: number; group: string;
}
export interface MarketGroup { name: string; items: ExtendedMarketItem[]; }
export interface ExtendedMarkets {
  groups: MarketGroup[];
  topEvBets: ExtendedMarketItem[];

  htft: Record<string, number>;

// ---- AH helper: integer line carries the push (v8.7 fix) ----
const makeAH = (diff: Map<number, number>) => {
  const gt = (t: number) => { let s = 0; diff.forEach((p, d) => { if (d > t) s += p; }); return s; };
  const lt = (t: number) => { let s = 0; diff.forEach((p, d) => { if (d < t) s += p; }); return s; };
  const eq = (t: number) => diff.get(t) ?? 0;
  const side = (M: number, dir: 'gt' | 'lt') => {
    const P = dir === 'gt' ? gt : lt;
    if (Number.isInteger(M)) return P(M) + 0.5 * eq(M);
    if (Math.abs(Math.abs(M) % 1 - 0.5) < 1e-9) return P(M); // half line: no push
    const I = Math.round(M);                                  // quarter line: integer neighbor push
    return P(I) + 0.5 * eq(I);
  };
  return { home: (L: number) => side(-L, 'gt'), away: (L: number) => side(-L, 'lt') };
};

// ---- HT/FT: copula MC joint + IPF to exact analytic margins ----
export function htftJointExact(lH: number, lA: number, rho: number, theta = 0.65): Record<string, number> {
  const fs = buildScoreMatrix(lH, lA, rho);
  const ft = outcomeProbs(fs.raw, fs.sum);
  const hs = buildScoreMatrix(lH / 2, lA / 2, rho);
  const ht = outcomeProbs(hs.raw, hs.sum);
  const mc = htftCopulaJoint(lH, lA, rho, theta, 20000, 7).combos;
  const K = ['1', 'X', '2'];
  const M = K.map((hk) => K.map((fk) => Math.max(mc[`${hk}_${fk}`] ?? 0.01, 1e-9)));
  const rowT = [ht.home, ht.draw, ht.away];
  const colT = [ft.home, ft.draw, ft.away];
  for (let it = 0; it < 30; it++) {          // iterative proportional fitting
    for (let r = 0; r < 3; r++) {
      const s = M[r][0] + M[r][1] + M[r][2];
      for (let c = 0; c < 3; c++) M[r][c] *= rowT[r] / s;
    }
    for (let c = 0; c < 3; c++) {
      const s = M[0][c] + M[1][c] + M[2][c];
      for (let r = 0; r < 3; r++) M[r][c] *= colT[c] / s;
    }
  }
  const out: Record<string, number> = {};
  K.forEach((hk, r) => K.forEach((fk, c) => { out[`${hk}_${fk}`] = M[r][c]; }));
  return out;
}

export function computeExtendedMarketsV9(
  matrix: ScoreEntry[], lH: number, lA: number, rho: number,
  userOdds: Record<string, number>,
): ExtendedMarkets {
  const P: number[][] = Array.from({ length: 10 }, () => new Array(10).fill(0));
  for (const e of matrix) P[e.i][e.j] = e.p;
  const flat: Record<string, number> = {};
  const groups: MarketGroup[] = [];

  const sumWhere = (f: (i: number, j: number) => boolean) => {
    let s = 0;
    for (let i = 0; i < 10; i++) for (let j = 0; j < 10; j++) if (f(i, j)) s += P[i][j];
    return s;
  };

  const mk = (key: string, label: string, prob: number, group: string): ExtendedMarketItem => {
    const odds = userOdds[key] && userOdds[key] > 1 ? userOdds[key] : 0;
    const item = { key, label, prob, odds, ev: odds > 1 ? prob * odds - 1 : 0, group };
    flat[key] = prob;
    return item;
  };

  // goal-difference distribution (for AH)
  const diffM = new Map<number, number>();
  for (let i = 0; i < 10; i++) for (let j = 0; j < 10; j++)
    diffM.set(i - j, (diffM.get(i - j) ?? 0) + P[i][j]);
  const ahFT = makeAH(diffM);

  const pHome = sumWhere((i, j) => i > j);
  const pDraw = sumWhere((i, j) => i === j);
  const pAway = 1 - pHome - pDraw;

  // 1. BTTS (audit fix: DC-corrected sum)
  const pBttsYes = sumWhere((i, j) => i > 0 && j > 0);
  groups.push({ name: 'BTTS', items: [
    mk('bttsYes', 'BTTS: Yes', pBttsYes, 'BTTS'),
    mk('bttsNo', 'BTTS: No', 1 - pBttsYes, 'BTTS'),
  ]});

  // 2. Over/Under FT
  const ouItems: ExtendedMarketItem[] = [];
  for (const line of [0.5, 1.5, 2.5, 3.5]) {
    const over = sumWhere((i, j) => i + j > line);
    const lk = String(line).replace('.', '');
    ouItems.push(mk(`over${lk}`, `Over ${line}`, over, 'O/U FT'));
    ouItems.push(mk(`under${lk}`, `Under ${line}`, 1 - over, 'O/U FT'));
  }
  groups.push({ name: 'Over/Under FT', items: ouItems });

  // 3. Exact goals
  const exItems: ExtendedMarketItem[] = [];
  for (let n = 0; n <= 5; n++)
    exItems.push(mk(`exact${n}`, `Exactly ${n} goals`, sumWhere((i, j) => i + j === n), 'Exact Goals'));
  exItems.push(mk('exact6plus', '6+ goals', sumWhere((i, j) => i + j >= 6), 'Exact Goals'));
  groups.push({ name: 'Exact Goals', items: exItems });

  // 4. Odd/Even
  const pOdd = sumWhere((i, j) => (i + j) % 2 === 1);
  groups.push({ name: 'Odd/Even', items: [
    mk('oddGoals', 'Odd total goals', pOdd, 'Odd/Even'),
    mk('evenGoals', 'Even total goals', 1 - pOdd, 'Odd/Even'),
  ]});

  // 5. Goal groups
  const range = (lo: number, hi: number) => sumWhere((i, j) => i + j >= lo && i + j <= hi);
  groups.push({ name: 'Goal Groups', items: [
    mk('group01', '0-1 goals', range(0, 1), 'Goal Groups'),
    mk('group23', '2-3 goals', range(2, 3), 'Goal Groups'),
    mk('group45', '4-5 goals', range(4, 5), 'Goal Groups'),
    mk('group6p', '6+ goals', sumWhere((i, j) => i + j >= 6), 'Goal Groups'),
  ]});

  // 6. Double chance
  const p1X = pHome + pDraw;
  groups.push({ name: 'Double Chance', items: [
    mk('dc1X', '1X (Home or Draw)', p1X, 'Double Chance'),
    mk('dcX2', 'X2 (Draw or Away)', pDraw + pAway, 'Double Chance'),
    mk('dc12', '12 (Home or Away)', pHome + pAway, 'Double Chance'),
  ]});

  // 7. Asian Handicap FT
  const ahLines = [-1, -0.75, -0.5, -0.25, 0, 0.25, 0.5, 0.75, 1];
  groups.push({ name: 'Asian Handicap FT', items: ahLines.map((L) =>
    mk(`ah${L}`, `AH Home ${L > 0 ? '+' : ''}${L}`, ahFT.home(L), 'Asian Handicap FT')) });

  // HT score matrix (lambda/2 per team, same rho)
  const hts = buildScoreMatrix(lH / 2, lA / 2, rho);
  const htp: number[][] = Array.from({ length: 10 }, () => new Array(10).fill(0));
  for (const e of hts.matrix) htp[e.i][e.j] = e.p;
  const htSumWhere = (f: (i: number, j: number) => boolean) => {
    let s = 0;
    for (let i = 0; i < 10; i++) for (let j = 0; j < 10; j++) if (f(i, j)) s += htp[i][j];
    return s;
  };
  const ht3 = outcomeProbs(hts.raw, hts.sum);
  const htDiff = new Map<number, number>();
  for (let i = 0; i < 10; i++) for (let j = 0; j < 10; j++)
    htDiff.set(i - j, (htDiff.get(i - j) ?? 0) + htp[i][j]);
  const ahHT = makeAH(htDiff);

  // 8. AH HT
  groups.push({ name: 'Asian Handicap HT', items: [
    mk('ahht-0.5', 'AH HT Home -0.5', ahHT.home(-0.5), 'Asian Handicap HT'),
    mk('ahht-0.25', 'AH HT Home -0.25', ahHT.home(-0.25), 'Asian Handicap HT'),
    mk('ahht0', 'AH HT Home 0', ahHT.home(0), 'Asian Handicap HT'),
  ]});

  // 9 & 10. Team goals O/U
  const teamItems: ExtendedMarketItem[] = [];
  for (const side of ['home', 'away'] as const) {
    for (const line of [0.5, 1.5, 2.5]) {
      let over = 0;
      for (let i = 0; i < 10; i++) for (let j = 0; j < 10; j++) {
        const g = side === 'home' ? i : j;
        if (g > line) over += P[i][j];
      }
      const lk = String(line).replace('.', '');
      teamItems.push(mk(`${side}Over${lk}`, `${side === 'home' ? 'Home' : 'Away'} Over ${line}`, over, 'Team Goals'));
      teamItems.push(mk(`${side}Under${lk}`, `${side === 'home' ? 'Home' : 'Away'} Under ${line}`, 1 - over, 'Team Goals'));
    }
  }
  groups.push({ name: 'Team Goals O/U', items: teamItems });

  // 11. First to score (lambda-ratio proportional)
  const pNeither = P[0][0];
  const pFtsH = (lH / (lH + lA)) * (1 - pNeither);
  const pFtsA = 1 - pNeither - pFtsH;
  groups.push({ name: 'First To Score', items: [
    mk('ftsHome', 'Home scores first', pFtsH, 'First To Score'),
    mk('ftsAway', 'Away scores first', pFtsA, 'First To Score'),
    mk('ftsNeither', 'Neither scores', pNeither, 'First To Score'),
  ]});

  // 12. 3-Way Half Time
  groups.push({ name: 'Half Time 3-Way', items: [
    mk('ht3Home', 'Home leading at HT', ht3.home, 'Half Time 3-Way'),
    mk('ht3Draw', 'Draw at HT', ht3.draw, 'Half Time 3-Way'),
    mk('ht3Away', 'Away leading at HT', ht3.away, 'Half Time 3-Way'),
  ]});

  // 13. BTTS HT
  const pBttsHT = htSumWhere((i, j) => i > 0 && j > 0);
  groups.push({ name: 'BTTS Half Time', items: [
    mk('bttsHTYes', 'BTTS at HT: Yes', pBttsHT, 'BTTS Half Time'),
    mk('bttsHTNo', 'BTTS at HT: No', 1 - pBttsHT, 'BTTS Half Time'),
  ]});

  // 14. O/U Half Time
  const htOver05 = htSumWhere((i, j) => i + j > 0.5);
  const htOver15 = htSumWhere((i, j) => i + j > 1.5);
  groups.push({ name: 'Over/Under HT', items: [
    mk('htOver05', 'HT Over 0.5', htOver05, 'Over/Under HT'),
    mk('htUnder05', 'HT Under 0.5', 1 - htOver05, 'Over/Under HT'),
    mk('htOver15', 'HT Over 1.5', htOver15, 'Over/Under HT'),
    mk('htUnder15', 'HT Under 1.5', 1 - htOver15, 'Over/Under HT'),
  ]});

  // 15. Half with most goals (independent Poisson halves)
  const m = (lH + lA) / 2;
  let p1st = 0, pNeitherHalf = 0;
  for (let a = 0; a < 10; a++) for (let b = 0; b < 10; b++) {
    const pj = poisson(a, m) * poisson(b, m);
    if (a > b) p1st += pj; else if (a === b) pNeitherHalf += pj;
  }
  groups.push({ name: 'Half With Most Goals', items: [
    mk('halfMost1st', '1st half', p1st, 'Half With Most Goals'),
    mk('halfMost2nd', '2nd half', 1 - p1st - pNeitherHalf, 'Half With Most Goals'),
    mk('halfMostNeither', 'Equal', pNeitherHalf, 'Half With Most Goals'),
  ]});

  // 16. HT/FT Double Result (v9 copula + IPF exact margins)
  const htft = htftJointExact(lH, lA, rho);
  const htftItems = Object.entries(htft).map(([k, prob]) =>
    mk(`htft_${k}`, `HT/FT ${k.replace('_', '/')}`, prob, 'HT/FT'))
    .sort((a, b) => b.prob - a.prob);
  groups.push({ name: 'HT/FT Double Result', items: htftItems });

  // 17. Euro Handicap
  const ehLines = ['-1:0', '0:1', '1:0', '0:-1', '-2:0', '0:2', '1:1', '2:1', '1:2', '2:0', '0:-2'];
  const ehItems: ExtendedMarketItem[] = [];
  for (const line of ehLines) {
    const [hStr, aStr] = line.split(':');
    const adjH = parseFloat(hStr), adjA = parseFloat(aStr);
    let ph = 0, pd = 0, pa = 0;
    for (let i = 0; i < 10; i++) for (let j = 0; j < 10; j++) {
      const vh = i + adjH, va = j + adjA;
      if (vh > va) ph += P[i][j]; else if (vh === va) pd += P[i][j]; else pa += P[i][j];
    }
    ehItems.push(mk(`eh_${line}_1`, `EH ${line}: Home`, ph, 'Euro Handicap'));
    ehItems.push(mk(`eh_${line}_X`, `EH ${line}: Draw`, pd, 'Euro Handicap'));
    ehItems.push(mk(`eh_${line}_2`, `EH ${line}: Away`, pa, 'Euro Handicap'));
  }
  groups.push({ name: 'Euro Handicap', items: ehItems });

  // 18. Correct Score (top 8)
  groups.push({ name: 'Correct Score', items:
    [...matrix].slice(0, 8).map((e) => mk(`cs_${e.i}_${e.j}`, `Score ${e.i}-${e.j}`, e.p, 'Correct Score')) });

  // 19. FTR + O/U combos
  const ftrOf = (i: number, j: number) => (i > j ? '1' : i === j ? 'X' : '2');
  const ftrOUItems: ExtendedMarketItem[] = [];
  for (const ftr of ['1', 'X', '2']) {
    for (const [lk, line, over] of [['O15', 1.5, true], ['U15', 1.5, false], ['O25', 2.5, true], ['U25', 2.5, false]] as const) {
      const p = sumWhere((i, j) => ftrOf(i, j) === ftr && (i + j > line) === over);
      ftrOUItems.push(mk(`ftrOU_${ftr}_${lk}`, `${ftr} & ${over ? 'Over' : 'Under'} ${line}`, p, 'FTR + O/U'));
    }
  }
  groups.push({ name: 'FTR + Over/Under', items: ftrOUItems });

  // 20-23. FTR + BTTS combos
  const ftrBTTSItems: ExtendedMarketItem[] = [];
  for (const ftr of ['1', 'X', '2']) {
    for (const [bk, yes] of [['Yes', true], ['No', false]] as const) {
      const p = sumWhere((i, j) => ftrOf(i, j) === ftr && (i > 0 && j > 0) === yes);
      ftrBTTSItems.push(mk(`ftrBTTS_${ftr}_${bk}`, `${ftr} & BTTS ${yes ? 'Yes' : 'No'}`, p, 'FTR + BTTS'));
    }
  }
  groups.push({ name: 'FTR + BTTS', items: ftrBTTSItems });

  const topEvBets = groups.flatMap((g) => g.items)
    .filter((it) => it.odds > 1 && it.ev > 0)
    .sort((a, b) => b.ev - a.ev)
    .slice(0, 4);

  return {
    groups, topEvBets, flat,
    pHome, pDraw, pAway, pBttsYes,
    pOver25: sumWhere((i, j) => i + j > 2.5),
    p1X, htft,
  };
}

const idbStorage = {

    if (typeof indexedDB === 'undefined') return null;
    return (await idbGet<string>(name)) ?? null;

    if (typeof indexedDB === 'undefined') return;

export const useAtlasStore = create<AtlasState>()(
  persist(
    (set, get) => ({
      matches: [], results: {}, feedback: [], phi: 0.10,
      predictions: [],
      standardParams: defaultParams(), jackpotParams: defaultParams(),
      simResults: {}, parlaySim: null, recalibrationLog: [],
      apiPicks: [], apiPicksLoading: false, apiPicksError: null,
      simsRunning: false,

      addMatch: () => set((s) => ({ matches: [...s.matches, emptyMatch(crypto.randomUUID())] })),
      removeMatch: (id) => set((s) => ({ matches: s.matches.filter((m) => m.id !== id) })),
      updateMatch: (id, patch) => set((s) => ({
        matches: s.matches.map((m) => (m.id === id ? { ...m, ...patch } : m)),
      })),
      updateOdds: (id, key, value) => set((s) => ({
        matches: s.matches.map((m) => (m.id === id
          ? { ...m, userOdds: { ...m.userOdds, [key]: value } } : m)),
      })),
      toggleEngine: (id) => set((s) => ({
        matches: s.matches.map((m) => (m.id === id
          ? { ...m, gameType: m.gameType === 'standard' ? 'jackpot' : 'standard' } : m)),
      })),
      setPhi: (v) => set({ phi: v }),

      recordPrediction: (match, a) => set((s) => ({
        predictions: [
          ...s.predictions.filter((p) => p.matchId !== match.id),
          {
            id: crypto.randomUUID(), matchId: match.id, gameType: match.gameType,
            pick: a.pick, confidence: a.confidence, probs: { ...a.probs },
            sofTriggered: a.sofTriggered, margin: a.margin, timestamp: Date.now(),
          },
        ],
      })),

      submitResult: (id, outcome) => {
        set((s) => ({
          results: { ...s.results, [id]: outcome },
          predictions: s.predictions.map((p) =>
            p.matchId === id && !p.result ? { ...p, result: outcome, resolvedAt: Date.now() } : p),
        }));
        // Rule 4 + auto-recalibrate every 5th resolved result
        const s = get();
        const pred = s.predictions.find((p) => p.matchId === id);
        if (!pred) return;
        const n = s.predictions.filter((p) => p.gameType === pred.gameType && p.result).length;
        if (n >= MIN_RECALIBRATION_RESULTS && n % MIN_RECALIBRATION_RESULTS === 0)
          get().forceRecalibrate(pred.gameType);
      },

      forceRecalibrate: (gameType) => {
        const s = get();
        const params = gameType === 'standard' ? s.standardParams : s.jackpotParams;
        const { params: np, message } = recalibrate(params, s.predictions, gameType);
        set(gameType === 'standard' ? { standardParams: np } : { jackpotParams: np });
        set((st) => ({ recalibrationLog: [...st.recalibrationLog, { at: Date.now(), gameType, message }] }));
      },

      clearLearningData: () => set({
        predictions: [], recalibrationLog: [],
        standardParams: defaultParams(), jackpotParams: defaultParams(),
      }),

      // Rule 10.3: sequential, yield to event loop between matches
      runAllSims: async () => {
        set({ simsRunning: true });
        for (const m of get().matches) {
          const h = parseFloat(m.h), d = parseFloat(m.d), a = parseFloat(m.a);
          if (h > 1 && d > 1 && a > 1) {
            const rho = m.gameType === 'jackpot' ? DC_RHO_JACKPOT : DC_RHO_STANDARD;
            const cal = calibrateLambda(h, d, a, rho, m.prior);
            const sm = buildScoreMatrix(cal.lambdaH, cal.lambdaA, rho);
            const grid = outcomeProbs(sm.raw, sm.sum);
            const sim = adaptiveSimulate(cal.lambdaH, cal.lambdaA, rho, { gridProbs: grid });
            set((s) => ({ simResults: { ...s.simResults, [m.id]: sim } }));
          }
          await new Promise((r) => setTimeout(r, 0));
        }
        set({ simsRunning: false });
      },

      runParlaySim: () => {
        const legs = get().matches
          .filter((m) => m.gameType === 'jackpot' &&
            parseFloat(m.h) > 1 && parseFloat(m.d) > 1 && parseFloat(m.a) > 1)
          .map((m) => {
            const a = atlasAnalyseJackpotV9(m, get().phi);
            return { lH: a.calibration.lambdaH, lA: a.calibration.lambdaA, rho: DC_RHO_JACKPOT, pick: a.pick };
          });
        if (!legs.length) return;
        const r = simulateParlay(legs);
        set({ parlaySim: { ...r, legs: legs.length } });
      },

      fetchApiPicks: async () => {
        set({ apiPicksLoading: true, apiPicksError: null });

export interface ParsedOdds { odds: Record<string, number>; count: number; unmatched: string[]; }

export function parseSmartPaste(text: string): ParsedOdds {
  const odds: Record<string, number> = {};
  const unmatched: string[] = [];
  if (!text || !text.trim()) return { odds, count: 0, unmatched };

  const lines = text.split(/\n/).map((l) => l.trim()).filter(Boolean);
  let pendingLabel: string | null = null;

  for (const line of lines) {
    const m = line.match(/^(.+?)\s*[:\t]\s*(\d+\.?\d*)\s*$/) ||
              line.match(/^(.+?)\s{2,}(\d+\.?\d*)\s*$/);
    if (m) {
      const key = resolveKey(m[1]);
      if (key) { odds[key] = parseFloat(m[2]); continue; }
      unmatched.push(line);
      continue;
    }
    if (/^\d+\.\d+$/.test(line) && pendingLabel) {
      const key = resolveKey(pendingLabel);
      if (key) odds[key] = parseFloat(line); else unmatched.push(pendingLabel);
      pendingLabel = null;
      continue;
    }
    pendingLabel = line;
  }
  return { odds, count: Object.keys(odds).length, unmatched };
}

function resolveKey(label: string): string | null {
  const norm = label.toLowerCase().trim();
  if (LABEL_MAP[norm]) return LABEL_MAP[norm];
  if (HTFT_ALIASES[norm]) return HTFT_ALIASES[norm];
  const cs = norm.match(/^(\d)\s*[:\-]\s*(\d)$/);
  if (cs && +cs[1] <= 9 && +cs[2] <= 9) return `cs_${cs[1]}_${cs[2]}`;
  const ftr = norm.match(/^(1|x|2|home|draw|away)\s*(?:&|and)\s*(.+)$/);
  if (ftr) {
    const f = ftr[1] === 'home' ? '1' : ftr[1] === 'draw' ? 'X' : ftr[1] === 'away' ? '2' : ftr[1].toUpperCase();
    const rest = ftr[2].trim();
    const ou = rest.match(/^(over|under|o|u)\s*(\d\.5)$/);
    if (ou) {
      const over = ou[1].startsWith('o');
      return `ftrOU_${f}_${over ? 'O' : 'U'}${ou[2].replace('.', '')}`;
    }
    if (/^(gg|btts|yes)$/.test(rest)) return `ftrBTTS_${f}_Yes`;
    if (/^(ng|no)$/.test(rest)) return `ftrBTTS_${f}_No`;
  }
  return null;
}

export function classifyMarket(text: string): BetMarket {
  const t = text.toLowerCase();

  if (/correct score|\bcs\b|\d-\d/.test(t)) return 'correct_score';
  if (/first (goal)?scorer|to score first/.test(t)) return 'first_goalscorer';
  if (/exact.*goals|total goals exactly/.test(t)) return 'exact_goals';
  if (/clean sheet/.test(t)) return 'clean_sheet';
  if (/goal range|\b0-1 goals\b|\b2-3 goals\b/.test(t)) return 'goal_range';
  if (/(home|away).*(over|under).*2\.5/.test(t)) return t.includes('home') ? 'home_over25' : 'away_over25';
  if (/handicap|ah\b/.test(t) || /(^|\s)[+-]1(\s|$)/.test(t) || /[+-]0\.\d+/.test(t)) return 'asian_handicap';

  if (/half time result|\bht\b.*(home|draw|away)/.test(t)) return 'half_time_result';

export function parseBetSlip(text: string): ParsedBet[] {
  const bets: ParsedBet[] = [];
  if (!text.trim()) return bets;
  const lines = text.split('\n').map((l) => l.trim()).filter(Boolean);
  let current: Partial<ParsedBet> = {};

  for (const line of lines) {
    if (/multi bet|bet id|stake|total odds/i.test(line)) {
      if (current.teams && current.odds) bets.push(finalize(current));
      current = {};
      continue;
    }
    const result: ParsedBet['result'] = /✅|✔|won/i.test(line) ? 'won' : /❌|✘|lost/i.test(line) ? 'lost' : 'pending';
    const pickLine = line.match(/^(?:pick|selection)[:\s]+(.+?)(?:\s*\((\d+\.?\d*)\))?\s*([✅✔❌✘])?$/i);
    if (pickLine) {
      current.pick = pickLine[1].trim();
      if (pickLine[2]) current.odds = parseFloat(pickLine[2]);
      if (pickLine[3]) current.result = pickLine[3] === '❌' || pickLine[3] === '✘' ? 'lost' : 'won';
      current.market = classifyMarket(current.pick);
      continue;
    }
    const oddsLine = line.match(/odds[:\s]+(\d+\.?\d*)/i);
    if (oddsLine) { current.odds = parseFloat(oddsLine[1]); continue; }
    const scoreLine = line.match(/(?:ft|score)[:\s]+(\d+)\s*[-:]\s*(\d+)/i);
    if (scoreLine) { current.score = `${scoreLine[1]}-${scoreLine[2]}`; continue; }
    if (/(vs| v | - )/i.test(line) || /^[A-Z].+ [A-Z]/.test(line)) {
      if (current.teams && current.odds) { bets.push(finalize(current)); current = {}; }
      current.teams = line;
      if (!current.result) current.result = result;
      if (!current.market) current.market = classifyMarket(line);
    }
  }
  if (current.teams && current.odds) bets.push(finalize(current));
  return bets;
}

function finalize(c: Partial<ParsedBet>): ParsedBet {
  return {
    teams: c.teams ?? 'Unknown', market: c.market ?? 'other',
    pick: c.pick ?? '', odds: c.odds ?? 1, result: c.result ?? 'pending', score: c.score,
  };
}

export function computeMarketStats(bets: ParsedBet[]) {
  const map = new Map<BetMarket, { total: number; wins: number }>();
  for (const b of bets) {
    if (b.result === 'pending') continue;
    const e = map.get(b.market) ?? { total: 0, wins: 0 };
    e.total++;
    if (b.result === 'won') e.wins++;
    map.set(b.market, e);
  }
  return [...map.entries()]
    .map(([market, s]) => ({ market, ...s, hitRate: s.total ? s.wins / s.total : 0 }))
    .sort((a, b) => b.total - a.total);
}

// ============================================================
// Walk-forward calibration diagnostics: Brier, log-loss,
// reliability bins, per-outcome accuracy. Powers CalibrationPanel.
// ============================================================
import type { Outcome } from '../atlas-types';
import type { PredictionRecord } from '../learning/bayesian-learning';

export interface ReliabilityBin { lo: number; hi: number; model: number; actual: number; n: number; }
export interface CalibrationMetrics {
  n: number; brier: number; logLoss: number; accuracy: number;
  byOutcome: Record<Outcome, { n: number; accuracy: number }>;
  reliability: ReliabilityBin[];
}

export function evaluatePredictions(records: PredictionRecord[]): CalibrationMetrics {
  const resolved = records.filter((r) => r.result && r.probs);
  const empty: CalibrationMetrics = {
    n: 0, brier: 0, logLoss: 0, accuracy: 0,
    byOutcome: { home: { n: 0, accuracy: 0 }, draw: { n: 0, accuracy: 0 }, away: { n: 0, accuracy: 0 } },
    reliability: [],
  };
  if (resolved.length < 5) return empty; // same 5-result minimum as learning

  let brier = 0, logLoss = 0, hits = 0;
  const byOut: Record<Outcome, { n: number; hits: number }> = {
    home: { n: 0, hits: 0 }, draw: { n: 0, hits: 0 }, away: { n: 0, hits: 0 },
  };
  const bins: ReliabilityBin[] = Array.from({ length: 5 }, (_, k) => ({
    lo: k * 0.2, hi: (k + 1) * 0.2, model: 0, actual: 0, n: 0,
  }));

  for (const r of resolved) {
    const actual = r.result!;
    const p = r.probs;
    for (const k of ['home', 'draw', 'away'] as Outcome[])
      brier += (p[k] - (k === actual ? 1 : 0)) ** 2;
    logLoss += -Math.log(Math.max(p[actual], 1e-9));
    if (r.pick === actual) hits++;
    byOut[actual].n++;
    if (r.pick === actual) byOut[actual].hits++;
    const pk = p[r.pick];
    const b = bins[Math.min(4, Math.floor(pk * 5))];
    b.model += pk; b.actual += r.pick === actual ? 1 : 0; b.n++;
  }

  const n = resolved.length;
  return {
    n,
    brier: brier / n,
    logLoss: logLoss / n,
    accuracy: hits / n,
    byOutcome: {
      home: { n: byOut.home.n, accuracy: byOut.home.n ? byOut.home.hits / byOut.home.n : 0 },
      draw: { n: byOut.draw.n, accuracy: byOut.draw.n ? byOut.draw.hits / byOut.draw.n : 0 },
      away: { n: byOut.away.n, accuracy: byOut.away.n ? byOut.away.hits / byOut.away.n : 0 },
    },
    reliability: bins.map((b) => ({
      lo: b.lo, hi: b.hi,
      model: b.n ? b.model / b.n : 0,
      actual: b.n ? b.actual / b.n : 0,
      n: b.n,
    })),
  };
}
