// ============================================================
// Smart Paste parser v3 — bookmaker profiles (SportPesa / Betika).
// Learned from live SportPesa KE layout:
//  • bracketed AH rows: "MANCHESTER CITY [-0.50] 1.93"
//  • bracketed Euro handicap: "1 (0:3) 14.00"
//  • bare-number exact goals, "6 OR MORE", "0 OR 1 GOAL"
//  • "1ST/2ND/NEITHER", "ANY OTHER", HT correct score section
//  • Asian totals HT (0.75 / 1.00 lines)
// Rule 3 preserved: parses on EVERY keystroke.
// ============================================================

export interface ParseOpts { profile?: PasteProfile; home?: string; away?: string; }

export interface ParsedOdds { odds: Record<string, number>; count: number; unmatched: string[]; }

function normLabel(raw: string): string {
  let n = raw.toLowerCase().replace(/\s+/g, ' ').trim();
  n = n.replace(/\s*goals?\b/g, '');
  n = n.replace(/^(over|under)\s*(\d\.5)0$/, '$1 $2');
  n = n.replace(/^(o|u)\s*(\d\.5)$/, (_m, a, b) => (a === 'o' ? 'over ' : 'under ') + b);
  n = n.replace(/^(over|under)(\d\.5)$/, '$1 $2');
  return n;
}
const normName = (s: string) => s.toLowerCase().replace(/[^a-z0-9]+/g, '');

function sideOfTeam(label: string, home?: string, away?: string): 'home' | 'away' | null {
  const l = normName(label);
  if (home && normName(home).length >= 4 && l.includes(normName(home).slice(0, 6))) return 'home';
  if (away && normName(away).length >= 4 && l.includes(normName(away).slice(0, 6))) return 'away';
  return null;
}

function resolveKey(label: string): string | null {
  const norm = normLabel(label);
  if (LABEL_MAP[norm]) return LABEL_MAP[norm];
  if (HTFT_ALIASES[norm]) return HTFT_ALIASES[norm];
  const prefixed = norm.match(/^(?:ht\/ft|halftime\/fulltime|half time\/full time)\s*(.+)$/);
  if (prefixed && HTFT_ALIASES[prefixed[1]]) return HTFT_ALIASES[prefixed[1]];
  const cs = norm.match(/^(\d)\s*[:\-]\s*(\d)$/);
  if (cs && +cs[1] <= 9 && +cs[2] <= 9) return `cs_${cs[1]}_${cs[2]}`;
  const ftr = norm.match(/^(1|x|2|home|draw|away)\s*(?:&|and)\s*(.+)$/);
  if (ftr) {
    const f = ftr[1] === 'home' ? '1' : ftr[1] === 'draw' ? 'X' : ftr[1] === 'away' ? '2' : ftr[1].toUpperCase();
    const rest = ftr[2].trim();
    const ou = rest.match(/^(over|under|o|u)\s*(\d\.5)$/);
    if (ou) return `ftrOU_${f}_${ou[1].startsWith('o') ? 'O' : 'U'}${ou[2].replace('.', '')}`;
    if (/^(gg|btts|yes)$/.test(rest)) return `ftrBTTS_${f}_Yes`;
    if (/^(ng|no)$/.test(rest)) return `ftrBTTS_${f}_No`;
  }
  return null;
}

const athtKey = (dir: 'over' | 'under', t: number): string | null => {
  const suf = t === 0.75 ? '075' : t === 1 ? '100' : t === 0.5 ? '050' : null;
  return suf ? `atht${dir === 'over' ? 'Over' : 'Under'}${suf}` : null;
};

function resolveWithSection(sec: Section | null, label: string): string | null {
  if (!sec) return null;
  const l = normLabel(label);
  const isH = /^(home|1)$/.test(l), isD = /^(draw|x)$/.test(l), isA = /^(away|2)$/.test(l);
  switch (sec.id) {
    case 'btts':
      if (/^(yes|y|gg)$/.test(l)) return 'bttsYes';
      if (/^(no|n|ng)$/.test(l)) return 'bttsNo';
      return null;
    case 'ou': {
      const line = sec.line ?? l.match(/(\d\.5)/)?.[1];
      if (!line) return null;
      const k = line.replace('.', '');
      if (/^over$/.test(l)) return `over${k}`;
      if (/^under$/.test(l)) return `under${k}`;
      return null;
    }
    case 'ouht': {
      const m = l.match(/^(over|under)\s*(\d+(?:\.\d+)?)$/);
      if (!m) return null;
      const t = parseFloat(m[2]);
      if (t === 0.5) return m[1] === 'over' ? 'htOver05' : 'htUnder05';
      if (t === 1.5) return m[1] === 'over' ? 'htOver15' : 'htUnder15';
      return null;
    }
    case 'atht': {
      const m = l.match(/^(over|under)\s*(\d+(?:\.\d+)?)$/);
      if (!m) return null;
      return athtKey(m[1] as 'over' | 'under', parseFloat(m[2]));
    }
    case 'exact':
      if (/^\d$/.test(l)) return `exact${l}`;
      if (/^6 or more$/.test(l)) return 'exact6plus';
      return null;
    case 'groups':
      if (/^0 or 1/.test(l)) return 'group01';
      if (/^2 or 3/.test(l)) return 'group23';
      if (/^4 or 5/.test(l)) return 'group45';
      if (/^6 or more/.test(l)) return 'group6p';
      return null;
    case 'oe':
      if (/^odd$/.test(l)) return 'oddGoals';
      if (/^even$/.test(l)) return 'evenGoals';
      return null;
    case 'ht':
      if (isH) return 'ht3Home'; if (isD) return 'ht3Draw'; if (isA) return 'ht3Away';
      return null;
    case 'sh':
      if (isH) return 'sh3Home'; if (isD) return 'sh3Draw'; if (isA) return 'sh3Away';
      return null;
    case 'cs':
      if (isH || /home/.test(l)) return 'csHome';
      if (isA || /away/.test(l)) return 'csAway';
      return null;
    case 'hmost':
      if (/^1st/.test(l)) return 'halfMost1st';
      if (/^2nd/.test(l)) return 'halfMost2nd';
      if (/^neither|^equal/.test(l)) return 'halfMostNeither';
      return null;
    case 'htcs': {
      if (/^any other$/.test(l)) return 'htcs_any';
      const m = l.match(/^(\d):(\d)$/);
      if (m && +m[1] <= 2 && +m[2] <= 2) return `htcs_${m[1]}_${m[2]}`;
      return null;
    }
    case 'cscore':
      if (/^any other$/.test(l)) return 'cs_any';
      return null;
    case 'htft':
      return HTFT_ALIASES[l] ?? null;
    default:
      return null;
  }
}

export function parseSmartPaste(text: string, opts: ParseOpts = {}): ParsedOdds {
  const odds: Record<string, number> = {};
  const unmatched: string[] = [];
  if (!text || !text.trim()) return { odds, count: 0, unmatched };
  const { home, away } = opts;

  const lines = text.split(/\n/).map((l) => l.trim()).filter(Boolean);
  let section: Section | null = null;
  let pendingLabel: string | null = null;
  const gridSections = ['exact', 'groups', 'hmost', 'htcs', 'oe', 'ouht', 'atht', 'cscore', 'eh'];

  for (const line of lines) {
    // Betika winning margin: "2 By 1 5.60"
    const wmRow = line.match(/^(1|2)\s+by\s+(\d)(\+)?\s+(\d+\.\d+)$/i);
    if (wmRow) {
      const key = resolveKey(`${wmRow[1]} by ${wmRow[2]}${wmRow[3] ? '+' : ''}`);
      if (key) odds[key] = parseFloat(wmRow[4]);
      continue;
    }
    // Euro handicap bracketed: "1 (0:3) 14.00" / "X (0:1): 3.90"
    const ehRow = line.match(/^(1|X|2)\s*\((-?\d+):(-?\d+)\)\s*[:\s]\s*(\d+\.\d+)$/i);
    if (ehRow) {
      const out = ehRow[1].toUpperCase() === 'X' ? 'X' : eh[1] === '1' ? '1' : '2';
      odds[`eh_${ehRow[2]}:${ehRow[3]}_${out}`] = parseFloat(ehRow[4]);
      continue;
    }
    // SportPesa bracketed AH: "MANCHESTER CITY [-0.50] 1.93"
    const ahRow = line.match(/^(.+?)\s*\[([+-]?\d+(?:\.\d+)?)\]\s*[:\s]\s*(\d+\.\d+)$/);
    if (ahRow) {
      const L = parseFloat(ahRow[2]);
      const val = parseFloat(ahRow[3]);
      if (section?.id === 'ahht') {
        const k = `ahht${L}`;
        if (['ahht-0.5', 'ahht-0.25', 'ahht0'].includes(k)) odds[k] = val;
      } else {
        const side = sideOfTeam(ahRow[1], home, away);
        if (side === 'away') odds[`ah${-L}_a`] = val;
        else odds[`ah${L}`] = val;
      }
      continue;
    }
    // label <sep> odds
    const m = line.match(/^(.+?)\s*[:\t]\s*(\d+\.?\d*)\s*$/) ||
              line.match(/^(.+?)\s{2,}(\d+\.?\d*)\s*$/);
    if (m) {
      const key = resolveKey(m[1]) ?? resolveWithSection(section, m[1]);
      if (key) { odds[key] = parseFloat(m[2]); continue; }
      unmatched.push(line);
      continue;
    }
    // section header?
    const low = line.toLowerCase();
    const sec = SECTION_RE.find((s) => s.re.test(low));
    if (sec && !/\d\.\d+/.test(line)) {
      section = { id: sec.id, line: low.match(/(\d\.5)/)?.[1] };
      pendingLabel = null;
      continue;
    }
    // grid-style single-space rows inside known sections ("0 15.00", "1ST 2.95")
    if (section && gridSections.includes(section.id)) {
      const g = line.match(/^(.+?)\s+(\d+\.\d+)$/);
      if (g) {
        const key = resolveWithSection(section, g[1]) ?? resolveKey(g[1]);
        if (key) { odds[key] = parseFloat(g[2]); continue; }
      }
    }
    // vertical: odds-only line follows a label line
    if (/^\d+\.\d+$/.test(line) && pendingLabel) {
      const key = resolveKey(pendingLabel) ?? resolveWithSection(section, pendingLabel);
      if (key) odds[key] = parseFloat(line); else unmatched.push(pendingLabel);
      pendingLabel = null;
      continue;
    }
    pendingLabel = line;
  }
  return { odds, count: Object.keys(odds).length, unmatched };
}

'use client';
import { useMemo, useState } from 'react';
import { parseSmartPaste } from '@/lib/v9/parsers/odds-parser';
import type { PasteProfile } from '@/lib/v9/parsers/odds-parser';
import { useAtlasStore } from '@/lib/v9/store/atlas-store';

export default function SmartPasteOdds({ matchId, onClose }: { matchId: string; onClose: () => void }) {
  const [text, setText] = useState('');
  const [profile, setProfile] = useState<PasteProfile>('auto');
  const updateOdds = useAtlasStore((s) => s.updateOdds);
  const match = useAtlasStore((s) => s.matches.find((m) => m.id === matchId));

  // Rule 3: parse on EVERY keystroke
  const parsed = useMemo(
    () => parseSmartPaste(text, { profile, home: match?.home, away: match?.away }),
    [text, profile, match?.home, match?.away],
  );

  // v9.2 sanity gate: flag internally-impossible odds before they poison EV
  const warnings = useMemo(() => {
    const w: string[] = [];
    const imp = (o?: number) => (o && o > 1 ? 1 / o : 0);
    const o = parsed.odds;
    if (o.bttsYes && o.bttsNo) {
      const s = imp(o.bttsYes) + imp(o.bttsNo);
      if (s < 0.98 || s > 1.3) w.push(`BTTS Yes+No implied ${(s * 100).toFixed(0)}% (sane range 100–130%) — check these odds belong to this match.`);
    }
    if (o.over25 && o.under25) {
      const s = imp(o.over25) + imp(o.under25);
      if (s < 0.98 || s > 1.3) w.push(`Over+Under 2.5 implied ${(s * 100).toFixed(0)}% — check for a mis-paste.`);
    }
    const dOdds = parseFloat(match?.d ?? '');
    if (o.exact0 && dOdds > 1 && imp(o.exact0) > 1 / dOdds)
      w.push('0-goals price implies more 0-0 results than the draw price allows — likely wrong match or wrong market.');
    return w;
  }, [parsed, match]);

  const apply = () => {
    Object.entries(parsed.odds).forEach(([k, v]) => updateOdds(matchId, k, v));
    onClose();
  };

  const pill = (p: PasteProfile, label: string) => (
    <button onClick={() => setProfile(p)}
      className={`rounded-full px-2.5 py-0.5 text-[10px] font-bold border ${
        profile === p ? 'bg-blue-500/20 text-blue-300 border-blue-500/40' : 'bg-zinc-800 text-zinc-500 border-zinc-700'}`}>
      {label}
    </button>
  );

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/70 p-4" onClick={onClose}>
      <div className="w-full max-w-lg rounded-xl border border-zinc-700 bg-zinc-900 p-4" onClick={(e) => e.stopPropagation()}>
        <div className="mb-2 flex items-center justify-between">
          <h3 className="text-sm font-semibold text-zinc-200">Smart Paste Odds</h3>
          <span className={`rounded-full px-2.5 py-0.5 text-xs font-bold tabular-nums ${
            parsed.count > 0 ? 'bg-emerald-500/15 text-emerald-400' : 'bg-zinc-800 text-zinc-500'}`}>
            {parsed.count} market{parsed.count === 1 ? '' : 's'} detected
          </span>
        </div>

        <div className="mb-2 flex items-center gap-1.5">
          <span className="text-[10px] uppercase text-zinc-500">Source:</span>
          {pill('auto', 'Auto')}
          {pill('sportpesa', 'SportPesa')}
          {pill('betika', 'Betika')}
        </div>

        <div className="mb-2 rounded-lg border border-zinc-800 bg-zinc-950/60 p-2 text-[10px] leading-relaxed text-zinc-500">
          <div className="mb-0.5 text-sky-400">Paste the whole market block — headers included. Supported:</div>
          <div>GG: 1.80 · Over 2.5: 1.72 · MANCHESTER CITY [-0.50] 1.93 · 1 (0:3) 14.00</div>
          <div>1/1 3.40 · 2 By 1 5.60 · sections like TOTAL GOALS EXACTLY / CORRECT SCORE - FIRST HALF</div>
        </div>

        <textarea
          value={text} onChange={(e) => setText(e.target.value)}
          placeholder={'TOTAL GOALS OVER/UNDER - HALF TIME\nOver 0.5 1.25\nUnder 0.5 2.85\nHALF TIME/FULL TIME\n1/1: 2.55'}
          className="h-40 w-full rounded-lg border border-zinc-700 bg-zinc-950 p-2 font-mono text-xs text-zinc-300 focus:border-blue-500 focus:outline-none"
        />

        {warnings.length > 0 && (
          <div className="mt-2 space-y-1 rounded-lg border border-amber-500/30 bg-amber-500/10 p-2">
            {warnings.map((w, i) => <div key={i} className="text-[10px] text-amber-400">⚠ {w}</div>)}
          </div>
        )}
        {parsed.unmatched.length > 0 && (
          <div className="mt-1 text-[10px] text-zinc-500">{parsed.unmatched.length} unmatched line(s)</div>
        )}

        <div className="mt-3 flex justify-end gap-2">
          <button onClick={onClose} className="rounded-lg px-3 py-1.5 text-xs text-zinc-400 hover:bg-zinc-800">Cancel</button>
          <button onClick={apply} disabled={!parsed.count}
            className="rounded-lg bg-blue-600 px-3 py-1.5 text-xs font-semibold text-white disabled:opacity-40 hover:bg-blue-500">
            Parse & Apply {parsed.count} odds
          </button>
        </div>
      </div>
    </div>
  );
}
