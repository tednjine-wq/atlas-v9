import { useState } from 'react';
import { useAtlasStore } from '@/lib/v9/store/atlas-store';
import { emptyMatch } from '@/lib/v9/atlas-types';

export default function ScrapeButton() {
  const [url, setUrl] = useState('');
  const [busy, setBusy] = useState(false);
  const [msg, setMsg] = useState<string | null>(null);

  const scrape = async () => {
    if (!url.trim()) return;
    setBusy(true); setMsg(null);
    try {
      const res = await fetch('/api/scrape-sportpesa', {
        method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ url }),
      });
      const data = await res.json();
      if (!res.ok) throw new Error(data.error ?? 'scrape failed');
      const fixtures = data.fixtures ?? [];
      if (fixtures.length > 0) {
        useAtlasStore.setState((s) => ({
          matches: [
            ...s.matches,
            ...fixtures.map((f: { home?: string; away?: string; h?: number; d?: number; a?: number }) => ({
              ...emptyMatch(crypto.randomUUID()),
              home: String(f.home ?? 'Home'), away: String(f.away ?? 'Away'),
              h: String(f.h ?? ''), d: String(f.d ?? ''), a: String(f.a ?? ''),
            })),
          ],
        }));
      }
      setMsg(`Extracted ${fixtures.length} fixtures${fixtures.length === 0 ? ' — use Smart Paste instead' : ''}`);
    } catch (e) {
      setMsg(`Scrape failed: ${e instanceof Error ? e.message : 'error'}. Use Smart Paste instead.`);
    } finally { setBusy(false); }
  };

  return (
    <div className="flex items-center gap-2">
      <input value={url} onChange={(e) => setUrl(e.target.value)} placeholder="SportPesa URL…"
        className="w-56 rounded-lg border border-zinc-700 bg-zinc-900 px-2 py-1.5 text-xs text-zinc-300 focus:border-blue-500 focus:outline-none" />

        {busy ? 'Scraping…' : 'Scrape'}
      </button>

  const phi = useAtlasStore((s) => s.phi);

    const std = matches.filter((m) => m.gameType === 'standard' && parseFloat(m.h) > 1 && parseFloat(m.d) > 1 && parseFloat(m.a) > 1);
    let totalEV = 0, totalVar = 0, totalStake = 0;
    const dist = { home: 0, draw: 0, away: 0 };

        const a = atlasAnalyseV9(m, phi, standardParams);

// Live odds via The Odds API — GET cached 3h, respects free tier (500 req/mo)
import { NextResponse } from 'next/server';

export const runtime = 'nodejs';
const CACHE_TTL = 3 * 60 * 60 * 1000;
let cache: { at: number; picks: unknown[] } | null = null;

const SPORTS = ['soccer_epl', 'soccer_spain_la_liga', 'soccer_italy_serie_a',
  'soccer_germany_bundesliga', 'soccer_france_ligue_one', 'soccer_uefa_champs_league'];

interface OddsOutcome { name: string; price: number; }
interface OddsMarket { key: string; outcomes: OddsOutcome[]; }
interface OddsBook { home_team?: string; away_team?: string; markets?: OddsMarket[]; }
interface OddsEvent {
  id: string; home_team: string; away_team: string; sport_title: string;
  commence_time: string; bookmakers: { key: string; markets: OddsMarket[] }[];
}

export async function GET() {
  const key = process.env.ODDS_API_KEY;
  if (!key) return NextResponse.json({ picks: [], error: 'ODDS_API_KEY not set' }, { status: 500 });
  if (cache && Date.now() - cache.at < CACHE_TTL) return NextResponse.json({ picks: cache.picks, cached: true });

  const picks: object[] = [];
  for (const sport of SPORTS) {
    try {
      const res = await fetch(
        `https://api.the-odds-api.com/v4/sports/${sport}/odds?apiKey=${key}&regions=eu&markets=h2h&oddsFormat=decimal`,
      );
      if (!res.ok) continue;
      const events: OddsEvent[] = await res.json();
      for (const ev of events) {
        const book = ev.bookmakers?.[0];
        const mkt = book?.markets?.find((m) => m.key === 'h2h');
        if (!mkt) continue;
        const o: Record<string, number> = {};
        for (const out of mkt.outcomes) o[out.name] = out.price;
        const h = o[ev.home_team], a = o[ev.away_team], d = o['Draw'];
        if (!h || !a || !d) continue;
        const bestHome = Math.max(...ev.bookmakers.map((b) =>
          b.markets.find((m) => m.key === 'h2h')?.outcomes.find((x) => x.name === ev.home_team)?.price ?? h));
        picks.push({
          id: ev.id, home: ev.home_team, away: ev.away_team,
          league: ev.sport_title, kickoff: ev.commence_time,
          h: String(h), d: String(d), a: String(a),
          bookmakers: ev.bookmakers.length,
          valueGap: bestHome / h - 1,
        });
      }
    } catch { /* skip failing sport */ }
  }
  cache = { at: Date.now(), picks };
  return NextResponse.json({ picks, cached: false });
}

// SportPesa scraper — 3 strategies: embedded JSON, DOM patterns, regex fallback
import { NextResponse } from 'next/server';

export const runtime = 'nodejs';

interface Fixture { home: string; away: string; h: number; d: number; a: number; }

export async function POST(req: Request) {
  try {
    const { url } = await req.json();
    if (!url || !/^https?:\/\//.test(url)) return NextResponse.json({ error: 'invalid url' }, { status: 400 });
    const res = await fetch(url, { headers: { 'User-Agent': 'Mozilla/5.0' } });
    const html = await res.text();

    // Strategy 1: embedded JSON
    const jsonMatch = html.match(/window\.__NUXT__\s*=\s*(\{.+?\});<\/script>/s) ||
                      html.match(/<script[^>]*type="application\/json"[^>]*>(\{.+?\})<\/script>/s);
    if (jsonMatch) {
      try {
        const data = JSON.parse(jsonMatch[1]);
        const fixtures = extractFixtures(data);
        if (fixtures.length) return NextResponse.json({ fixtures, strategy: 'json' });
      } catch { /* fall through */ }
    }
    // Strategy 2/3: regex over HTML
    const fixtures: Fixture[] = [];
    const rowRe = /data-home="([^"]+)"[^>]*data-away="([^"]+)"[^>]*data-odds="([\d.,]+)"/g;
    let m;
    while ((m = rowRe.exec(html)) !== null) {
      const odds = m[3].split(',').map(parseFloat);
      if (odds.length === 3) fixtures.push({ home: m[1], away: m[2], h: odds[0], d: odds[1], a: odds[2] });
    }
    return NextResponse.json({ fixtures, strategy: fixtures.length ? 'regex' : 'none' });
  } catch (e) {
    return NextResponse.json({ error: e instanceof Error ? e.message : 'scrape failed' }, { status: 500 });
  }
}

function extractFixtures(obj: unknown, acc: Fixture[] = []): Fixture[] {
  if (!obj || typeof obj !== 'object') return acc;
  if (Array.isArray(obj)) { obj.forEach((x) => extractFixtures(x, acc)); return acc; }
  const rec = obj as Record<string, unknown>;
  if (rec.homeTeam && rec.awayTeam && rec.odds) {
    const o = rec.odds as Record<string, unknown>;
    acc.push({
      home: String(rec.homeTeam), away: String(rec.awayTeam),
      h: Number(o.home ?? 0), d: Number(o.draw ?? 0), a: Number(o.away ?? 0),
    });
  }
  Object.values(rec).forEach((v) => extractFixtures(v, acc));
  return acc.slice(0, 50);
}
