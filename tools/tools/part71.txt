// ============================================================
// Smart Paste v10 — Betika dialect COMPLETED.
// v10 adds: full skip table for unmodeled Betika noise
// (corners/bookings/scorers/multigoals/margins/combos), ordered
// alias table so Betika titles resolve to the same section IDs
// as SportPesa, and dynamic team-section routing
// ("<Home> Total" → homeOU, other "<Team> …" → skip).
// One screener, two dialects — math stays bookmaker-agnostic.
// ============================================================
export type PasteProfile = 'auto' | 'sportpesa' | 'betika';
export interface ParseOpts { profile?: PasteProfile; home?: string; away?: string; }
export interface ParsedOdds {
  odds: Record<string, number>; count: number; unmatched: string[];
  main?: { h: number; d: number; a: number };
}

const norm = (s: string) => s.toLowerCase().replace(/\s+/g, ' ').trim();
const cleanLine = (s: string) => s.replace(/\|/g, ' ').trim();

const LABEL_LIKE = /^(over|under)\s|^(yes|no|gg|ng)$|^(1|x|2)\s*(and|&)\s|^(1 or x|x or 2|1 or 2)$|^\d+ or \d+ goal|^6 or more$|^\d+:\d+$|^any other$|^(home|away|draw|neither|odd|even|1st|2nd)$|^[a-z0-9'. -]+\[[+-]?\d+\.\d+\]$|^(1|x|2)\s\(|^\d$|^x$|^(1x|x2|12|1\/x|x\/2|1\/2)$|^(x\/x|1\/1|2\/2|1\/x|1\/2|2\/1|x\/1|x\/2)$|^6\+$/i;

// ORDER MATTERS: skips & specific headers first, general last.

  { re: /10 minutes/, id: 'skip' },
  { re: /multigoals/, id: 'skip' },
  { re: /winning margin/, id: 'skip' },
  { re: /goalscorer/, id: 'skip' },
  { re: /corner/, id: 'skip' },
  { re: /booking/, id: 'skip' },
  { re: /clean sheet/, id: 'skip' },
  { re: /win to nil/, id: 'skip' },
  { re: /win both halves|win either half|score in both halves/, id: 'skip' },
  { re: /both halves (over|under)/, id: 'skip' },
  { re: /1st\/2nd half/, id: 'skip' },
  { re: /who will win/, id: 'skip' },
  { re: /total & both teams/, id: 'skip' },
  { re: /double chance &/, id: 'skip' },
  { re: /halftime\/fulltime &/, id: 'skip' },
  { re: /2nd half/, id: 'skip' },
  { re: /1st half - 1x2 &/, id: 'skip' },
  { re: /1st half - double chance/, id: 'skip' },
  { re: /1st half - handicap/, id: 'skip' },
  { re: /1st half - exact goals/, id: 'skip' },
  { re: /asian handicap - half time/, id: 'skip' },
  { re: /draw no bet/, id: 'skip' },
  // ---- modeled markets ----
  { re: /first team to score|1st goal/, id: 'fts' },
  { re: /1x2 & both teams|result\s*\+\s*both teams/, id: 'ftrBTTS' },
  { re: /1x2 & total|result\s*\+\s*total goals/, id: 'ftrOU' },
  { re: /1st half - both teams to score|both teams to score\s*-\s*first half/, id: 'bttsHT' },
  { re: /both teams to score/, id: 'btts' },
  { re: /1st half - 1x2|3 way\s*-\s*first half/, id: 'ht3' },
  { re: /3 way/, id: 'main' },
  { re: /double chance/, id: 'dc' },
  { re: /asian total goals over\/under\s*-\s*half time/, id: 'athtHT' },
  { re: /asian total goals/, id: 'skip' },
  { re: /1st half - total|over\/under\s*-\s*half time/, id: 'ouHT' },
  { re: /over\/under\s+home\s+team|home\s+team\s*-\s*full\s+time/, id: 'homeOU' },
  { re: /over\/under\s+away\s+team|away\s+team\s*-\s*full\s+time/, id: 'awayOU' },

  { re: /^total$/, id: 'ouFT' },
  { re: /total goals exactly|exact goals/, id: 'exact' },
  { re: /number of goals in groups/, id: 'groups' },
  { re: /correct score\s*-\s*first half|1st half - correct score/, id: 'htcs' },
  { re: /correct score/, id: 'cs' },
  { re: /half with most goals/, id: 'hmost' },
  { re: /half time\/full time|halftime\/fulltime/, id: 'htft' },

const HTFT_ALIASES: Record<string, string> = {
  '1/1': 'htft_1_1', '1/x': 'htft_1_X', '1/2': 'htft_1_2',
  'x/1': 'htft_X_1', 'x/x': 'htft_X_X', 'x/2': 'htft_X_2',
  '2/1': 'htft_2_1', '2/x': 'htft_2_X', '2/2': 'htft_2_2',
};

function ouKey(dir: 'over' | 'under', line: number, prefix: '' | 'ht' | 'home' | 'away' | 'atht'): string | null {
  const map: Record<string, string> = { '0.5': '05', '1.5': '15', '2.5': '25', '3.5': '35', '4.5': '45', '0.75': '075', '1': '100', '1.0': '100', '1.00': '100' };
  const allowed: Record<string, string[]> = {
    '': ['05', '15', '25', '35', '45'],
    ht: ['05', '15'],
    home: ['05', '15', '25', '35'],
    away: ['05', '15', '25', '35'],
    atht: ['075', '100'],
  };
  const k = map[String(line)];
  if (!k || !allowed[prefix].includes(k)) return null;
  return `${prefix}${dir === 'over' ? 'Over' : 'Under'}${k}`;
}

function resolveIn(sec: string, label: string, rowInSec: number): string | null {
  const l = norm(label);
  const ou = l.match(/^(over|under)\s*(\d+(?:\.\d+)?)$/);
  const isD = /^(draw|x)$/.test(l);
  switch (sec) {
    case 'btts':
      if (/\band\b|\d/.test(l)) return null;
      return /yes|gg/.test(l) ? 'bttsYes' : /no|ng/.test(l) ? 'bttsNo' : null;
    case 'bttsHT': return /yes|gg/.test(l) ? 'bttsHTYes' : /no|ng/.test(l) ? 'bttsHTNo' : null;
    case 'dc':
      if (/1 or x|^1x$|^1\/x$/.test(l)) return 'dc1X';
      if (/x or 2|^x2$|^x\/2$/.test(l)) return 'dcX2';
      if (/1 or 2|^12$|^1\/2$/.test(l)) return 'dc12';
      return null;
    case 'groups':
      if (/^0 or 1/.test(l)) return 'group01';
      if (/^2 or 3/.test(l)) return 'group23';
      if (/^4 or 5/.test(l)) return 'group45';
      if (/^6 or more/.test(l)) return 'group6p';
      return null;
    case 'exact':
      if (/^6 or more|^6\+$/.test(l)) return 'exact6plus';
      if (/^\d$/.test(l)) return `exact${l}`;
      return null;
    case 'ouFT':
      if (/\band\b/.test(l)) return null;
      return ou ? ouKey(ou[1] as 'over' | 'under', parseFloat(ou[2]), '') : null;
    case 'ouHT': return ou ? ouKey(ou[1] as 'over' | 'under', parseFloat(ou[2]), 'ht') : null;
    case 'homeOU': return ou ? ouKey(ou[1] as 'over' | 'under', parseFloat(ou[2]), 'home') : null;
    case 'awayOU': return ou ? ouKey(ou[1] as 'over' | 'under', parseFloat(ou[2]), 'away') : null;
    case 'athtHT': return ou ? ouKey(ou[1] as 'over' | 'under', parseFloat(ou[2]), 'atht') : null;
    case 'ahFT': {
      const m = l.match(/\[\s*([+-]?\d+(?:\.\d+)?)\s*\]/);
      if (!m) return null;
      const pos = m[1].startsWith('+');
      const lk = String(Math.abs(parseFloat(m[1])));
      return pos ? `ah${lk}` : `ah${lk}_a`;
    }
    case 'fts':
      if (/neither/.test(l)) return 'ftsNeither';
      if (/^1$|home/.test(l)) return 'ftsHome';
      if (/^2$|away/.test(l)) return 'ftsAway';
      return null;
    case 'ht3':
      if (isD) return 'ht3Draw';
      return rowInSec === 0 ? 'ht3Home' : 'ht3Away';
    case 'hmost':
      if (/^1st/.test(l)) return 'halfMost1st';
      if (/^2nd/.test(l)) return 'halfMost2nd';
      if (/neither|equal/.test(l)) return 'halfMostNeither';
      return null;
    case 'htft': return HTFT_ALIASES[l] ?? null;
    case 'eh': {
      const m = l.match(/^(1|x|2)\s*\(\s*(-?\d+)\s*:\s*(-?\d+)\s*\)$/);
      if (!m) return null;
      const out = m[1] === 'x' ? 'X' : m[1] === '1' ? '1' : '2';
      return `eh_${m[2]}:${m[3]}_${out}`;
    }
    case 'cs':
    case 'htcs': {
      const pre = sec === 'cs' ? 'cs' : 'htcs';
      if (/any other/.test(l)) return `${pre}_any`;
      const m = l.match(/^(\d+)\s*[:\-]\s*(\d+)$/);
      if (m && +m[1] <= 9 && +m[2] <= 9) return `${pre}_${m[1]}_${m[2]}`;
      return null;
    }
    case 'ftrOU': {
      const m = l.match(/^(1|x|2)\s*(?:&|and)\s*(over|under)\s*(\d\.5)$/);
      if (!m) return null;
      const f = m[1] === 'x' ? 'X' : m[1];
      return `ftrOU_${f}_${m[2] === 'over' ? 'O' : 'U'}${m[3].replace('.', '')}`;
    }
    case 'ftrBTTS': {
      const m = l.match(/^(1|x|2)\s*(?:&|and)\s*(yes|no)/);
      if (!m) return null;
      const f = m[1] === 'x' ? 'X' : m[1];
      return `ftrBTTS_${f}_${m[2] === 'yes' ? 'Yes' : 'No'}`;
    }
    case 'oe': return /^odd/.test(l) ? 'oddGoals' : /^even/.test(l) ? 'evenGoals' : null;
    default: return null;
  }
}

const LABEL_MAP: Record<string, string> = {
  'gg': 'bttsYes', 'btts': 'bttsYes', 'btts yes': 'bttsYes', 'yes (gg)': 'bttsYes',
  'ng': 'bttsNo', 'btts no': 'bttsNo', 'no (ng)': 'bttsNo',
  'over 0.5': 'over05', 'under 0.5': 'under05', 'over 1.5': 'over15', 'under 1.5': 'under15',
  'over 2.5': 'over25', 'under 2.5': 'under25', 'over 3.5': 'over35', 'under 3.5': 'under35',
  '1x': 'dc1X', 'x2': 'dcX2', '12': 'dc12', '1 or x': 'dc1X', 'x or 2': 'dcX2', '1 or 2': 'dc12',
  'odd': 'oddGoals', 'even': 'evenGoals',
  '0 or 1 goal': 'group01', '2 or 3 goals': 'group23', '4 or 5 goals': 'group45', '6 or more': 'group6p',
};

export function parseSmartPaste(text: string, opts: ParseOpts = {}): ParsedOdds {
  const odds: Record<string, number> = {};
  const unmatched: string[] = [];
  let main: { h: number; d: number; a: number } | undefined;
  if (!text || !text.trim()) return { odds, count: 0, unmatched };

  const raw = text.split(/\n/).map(cleanLine).filter(Boolean);

  const lines: string[] = [];
  for (let i = 0; i < raw.length; i++) {
    const cur = raw[i], nxt = raw[i + 1];
    if (nxt && /^\d+\.\d+$/.test(nxt) && (/^(over|under)\s/i.test(cur) || /\[[+-]?\d+(?:\.\d+)?\]$/.test(cur))) {
      lines.push(`${cur} : ${nxt}`);
      i++;
    } else lines.push(cur);
  }

  let sec: string | null = null;
  let rowInSec = 0;
  let pending: string | null = null;

  const homeTok = opts.home?.toLowerCase().split(' ').filter(w => w.length > 3).pop() ?? '';
  const awayTok = opts.away?.toLowerCase().split(' ').filter(w => w.length > 3).pop() ?? '';

  const sink = (label: string, val: number) => {
    if (sec === 'skip') return;
    if (sec === 'main') {
      const l = norm(label);
      const cur = main ?? { h: 0, d: 0, a: 0 };
      if (/draw|x/.test(l)) cur.d = val;
      else if (/away|2/.test(l) || (cur.h && cur.d && !cur.a)) cur.a = val;
      else cur.h = val;
      main = cur;
      return;
    }
    if (sec) {
      const key = resolveIn(sec, label, rowInSec);
      if (key) { odds[key] = val; rowInSec++; } else unmatched.push(`${label} ${val}`);
      return;
    }
    const l = norm(label);
    const key = LABEL_MAP[l] ?? HTFT_ALIASES[l] ?? null;
    if (key) odds[key] = val; else unmatched.push(`${label} ${val}`);
  };

  for (const line of lines) {
    if (/\d{2}-\d{2}-\d{4}/.test(line)) continue;
    const low = norm(line);

    // v10: dynamic team sections — "<Team> Total" (FT) maps to
    // home/away OU; every other team-specific header is skipped.
    const isHome = homeTok.length > 2 && low.includes(homeTok);
    const isAway = awayTok.length > 2 && low.includes(awayTok);
    if (isHome || isAway) {
      sec = (isHome || !isAway) && low.includes('total') && !low.includes('half') ? 'homeOU'
        : isAway && low.includes('total') && !low.includes('half') ? 'awayOU'
        : 'skip';
      rowInSec = 0; pending = null; continue;
    }

    const hdr = SECTION_RE.find((s) => s.re.test(low));
    if (hdr && !/\d\.\d{2}/.test(line)) { sec = hdr.id; rowInSec = 0; pending = null; continue; }

    const m = line.match(/^(.+?)\s*[:\t]\s*(\d+\.\d+)\s*$/) ||
              line.match(/^(.+?)\s{2,}(\d+\.\d+)\s*$/);
    if (m) { pending = null; sink(m[1], parseFloat(m[2])); continue; }

    if (!/:/.test(line) && LABEL_LIKE.test(line)) { pending = line; continue; }

    if (/^\d+\.\d+$/.test(line) && pending) { sink(pending, parseFloat(line)); pending = null; continue; }

    const ms = sec ? line.match(/^(.+?)\s+(\d+\.\d+)$/) : null;
    if (ms) { pending = null; sink(ms[1], parseFloat(ms[2])); continue; }

    pending = line;
  }
  if (main && !(main.h > 1 && main.d > 1 && main.a > 1)) main = undefined;
  return { odds, count: Object.keys(odds).length, unmatched, main };
}
