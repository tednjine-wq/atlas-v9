// ============================================================
// Poisson PMF with precomputed log-factorials (v8.7 exact spec)
// ============================================================

export const LF: number[] = (() => {
  const lf = new Array<number>(21).fill(0);
  for (let i = 1; i <= 20; i++) lf[i] = lf[i - 1] + Math.log(i);
  return lf;
})();

export function poisson(k: number, lam: number): number {
  if (k < 0 || k > 15 || lam <= 0) return 0;
  return Math.exp(-lam + k * Math.log(lam) - LF[k]);
}

// Inverse-normal CDF (Acklam) — needed by copula module
export function normPPF(p: number): number {
  const a = [-3.969683028665376e1, 2.209460984245205e2, -2.759285104469687e2,
             1.383577518672690e2, -3.066479806614716e1, 2.506628277459239];
  const b = [-5.447609879822406e1, 1.615858368580409e2, -1.556989798598866e2,
             6.680131188771972e1, -1.328068155288572e1];
  const c = [-7.784894002430293e-3, -3.223964580411365e-1, -2.400758277161838,
             -2.549732539343734, 4.374664141464968, 2.938163982698783];
  const d = [7.784695709041462e-3, 3.224671290700398e-1, 2.445134137142996,
             3.754408661907416];
  const plow = 0.02425;
  if (p < plow) {
    const q = Math.sqrt(-2 * Math.log(p));
    return (((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) /
           ((((d[0]*q+d[1])*q+d[2])*q+d[3])*q+1);
  }
  if (p > 1 - plow) {
    const q = Math.sqrt(-2 * Math.log(1 - p));
    return -(((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) /
            ((((d[0]*q+d[1])*q+d[2])*q+d[3])*q+1);
  }
  const q = p - 0.5, r = q * q;
  return (((((a[0]*r+a[1])*r+a[2])*r+a[3])*r+a[4])*r+a[5])*q /
         (((((b[0]*r+b[1])*r+b[2])*r+b[3])*r+b[4])*r+1);
}

// erf-based normal CDF
export function normCDF(x: number): number {
  return 0.5 * (1 + erf(x / Math.SQRT2));
}

function erf(x: number): number {
  const s = x < 0 ? -1 : 1;
  x = Math.abs(x);
  const t = 1 / (1 + 0.3275911 * x);
  const y = 1 - (((((1.061405429 * t - 1.453152027) * t) + 1.421413741) * t
            - 0.284496736) * t + 0.254829592) * t * Math.exp(-x * x);
  return s * y;
}

// Deterministic seeded RNG (mulberry32) — reproducibility in workers
export function mulberry32(seed: number): () => number {
  let s = seed >>> 0;
  return () => {
    s |= 0; s = (s + 0x6D2B79F5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// ============================================================
// Dixon-Coles tau + score matrix (v8.7 exact, v8.4 rho-consistency fix baked in)
// ============================================================
import { poisson } from './poisson';
import type { ScoreEntry, Outcome } from '../atlas-types';

export function dcTau(i: number, j: number, lH: number, lA: number, rho: number): number {
  if (i === 0 && j === 0) return 1 - lH * lA * rho;
  if (i === 0 && j === 1) return 1 + lH * rho;
  if (i === 1 && j === 0) return 1 + lA * rho;
  if (i === 1 && j === 1) return 1 - rho;
  return 1;
}

export interface ScoreMatrixResult {
  matrix: ScoreEntry[];       // sorted by p desc
  raw: Float64Array;          // flat 10x10, unsorted, row i col j
  sum: number;
}

export function buildScoreMatrix(lH: number, lA: number, rho: number): ScoreMatrixResult {
  const raw = new Float64Array(100);
  let sum = 0;
  for (let i = 0; i < 10; i++) {
    const pi = poisson(i, lH);
    for (let j = 0; j < 10; j++) {
      const p = dcTau(i, j, lH, lA, rho) * pi * poisson(j, lA);
      raw[i * 10 + j] = p;
      sum += p;
    }
  }
  const matrix: ScoreEntry[] = [];
  for (let i = 0; i < 10; i++)
    for (let j = 0; j < 10; j++)
      matrix.push({ s: `${i}:${j}`, i, j, p: raw[i * 10 + j] / sum });
  matrix.sort((a, b) => b.p - a.p);
  return { matrix, raw, sum };
}

export function outcomeProbs(raw: Float64Array, sum: number): Record<Outcome, number> {
  let h = 0, d = 0, a = 0;
  for (let i = 0; i < 10; i++)
    for (let j = 0; j < 10; j++) {
      const p = raw[i * 10 + j] / sum;
      if (i > j) h += p; else if (i === j) d += p; else a += p;
    }
  return { home: h, draw: d, away: a };
}

// ============================================================
// Lambda calibration: v8.7 coarse-fine grid search, enhanced with
// research-backed HYBRID BAYESIAN blending (market + strength prior,
// precision-weighted) [Frontiers 2026 ensemble / JRSS-C 2026 findings]
// ============================================================
import { buildScoreMatrix, outcomeProbs } from './dixon-coles';
import type { CalibrationResult, TeamStrengthPrior } from '../atlas-types';

// --- Margin stripping (Layer 1, v8.7 exact) ---
export function stripMargin(h: number, d: number, a: number) {
  const pH = 1 / h, pD = 1 / d, pA = 1 / a;
  const s = pH + pD + pA;
  return {
    probs: [pH / s, pD / s, pA / s] as [number, number, number],
    margin: s - 1,
  };
}

// --- v9.0 Hybrid Bayesian blend ---
// Market precision scales as 1/margin (tighter books = more information).
// Prior precision = pseudo-count from strength model (0..20).
// Blend is precision-weighted, then renormalized.
export function hybridBlend(
  marketProbs: [number, number, number],
  margin: number,
  prior: TeamStrengthPrior | undefined,
): { blended: [number, number, number]; hybrid: CalibrationResult['hybrid'] } {
  if (!prior || prior.confidence <= 0) {
    return {
      blended: marketProbs,
      hybrid: {
        source: 'market-only', marketProbs, priorProbs: null,
        blendedProbs: marketProbs, marketPrecision: 1 / Math.max(margin, 0.01), priorPrecision: 0,
      },
    };
  }
  const wM = 1 / Math.max(margin, 0.01);
  const wP = Math.min(prior.confidence, 20);
  const priorProbs: [number, number, number] = [prior.pHome, prior.pDraw, prior.pAway];
  const blended = marketProbs.map((pm, k) =>
    (wM * pm + wP * priorProbs[k]) / (wM + wP),
  ) as [number, number, number];
  const s = blended.reduce((x, y) => x + y, 0);
  const norm = blended.map((b) => b / s) as [number, number, number];
  return {
    blended: norm,
    hybrid: { source: 'hybrid', marketProbs, priorProbs, blendedProbs: norm, marketPrecision: wM, priorPrecision: wP },
  };
}

// --- Coarse-fine grid search (v8.7 spec: 0.3..3.5 step 0.15 coarse, ±0.15 step 0.02 fine) ---
export function calibrateLambda(
  hOdds: number, dOdds: number, aOdds: number,
  rho: number,
  prior?: TeamStrengthPrior,
): CalibrationResult {
  const { probs: marketProbs, margin } = stripMargin(hOdds, dOdds, aOdds);
  const { blended, hybrid } = hybridBlend(marketProbs, margin, prior);
  const [tH, tD, tA] = blended;

  const err = (lH: number, lA: number) => {
    const { raw, sum } = buildScoreMatrix(lH, lA, rho);
    const { home, draw, away } = outcomeProbs(raw, sum);
    return (home - tH) ** 2 + (draw - tD) ** 2 + (away - tA) ** 2;
  };

  // coarse pass
  let bestH = 1, bestA = 1, bestE = Infinity;
  for (let lH = 0.3; lH <= 3.5 + 1e-9; lH += 0.15)
    for (let lA = 0.3; lA <= 3.5 + 1e-9; lA += 0.15) {
      const e = err(lH, lA);
      if (e < bestE) { bestE = e; bestH = lH; bestA = lA; }
    }

  // fine pass (v8.4 fix: SAME rho as downstream — guaranteed by construction)
  for (let lH = Math.max(0.1, bestH - 0.15); lH <= bestH + 0.15 + 1e-9; lH += 0.02)
    for (let lA = Math.max(0.1, bestA - 0.15); lA <= bestA + 0.15 + 1e-9; lA += 0.02) {
      const e = err(lH, lA);
      if (e < bestE) { bestE = e; bestH = lH; bestA = lA; }
    }

  return { lambdaH: bestH, lambdaA: bestA, sse: bestE, hybrid };
}

// ============================================================
// Adaptive Monte Carlo — v9.0 upgrade over fixed 10k:
//  • CDF-inverse sampling (v8.7, ~100x faster than naive)
//  • ANTITHETIC VARIATES: sample u and 1-u as pairs → ~3x variance reduction
//  • SEQUENTIAL STOPPING: start 2k draws, extend to 20k only if
//    CI half-width > tolerance (PEMC-inspired, arXiv 2024)
//  • RULE 5 PRESERVED: MC is a cross-check, NEVER replaces grid model
// ============================================================
import { dcTau } from './dixon-coles';
import { poisson, mulberry32 } from './poisson';
import type { Outcome, SimResult, DisagreementLevel, SimRecommendation } from '../atlas-types';

interface Sampler { cdf: Float64Array; scores: Uint8Array; } // scores = i*10+j

export function buildSampler(lH: number, lA: number, rho: number): Sampler {
  const cdf = new Float64Array(100);
  const scores = new Uint8Array(100);
  let acc = 0, idx = 0;
  for (let i = 0; i < 10; i++) {
    const pi = poisson(i, lH);
    for (let j = 0; j < 10; j++) {
      acc += dcTau(i, j, lH, lA, rho) * pi * poisson(j, lA);
      cdf[idx] = acc;
      scores[idx] = i * 10 + j;
      idx++;
    }
  }
  for (let k = 0; k < 100; k++) cdf[k] /= acc; // normalize final value to exactly 1
  return { cdf, scores };
}

function sample(s: Sampler, u: number): number {
  let lo = 0, hi = 99;
  while (lo < hi) {
    const mid = (lo + hi) >> 1;
    if (s.cdf[mid] < u) lo = mid + 1; else hi = mid;
  }
  return s.scores[lo];
}

export interface AdaptiveMCOptions {
  minPairs?: number;    // default 1000 (=2k draws)
  maxPairs?: number;    // default 10000 (=20k draws)
  tol?: number;         // CI half-width target, default 0.008
  seed?: number;
  gridProbs?: Record<Outcome, number>;
  gridPick?: Outcome;
}

export function adaptiveSimulate(lH: number, lA: number, rho: number, opts: AdaptiveMCOptions = {}): SimResult {
  const minPairs = opts.minPairs ?? 1000;
  const maxPairs = opts.maxPairs ?? 10000;
  const tol = opts.tol ?? 0.008;
  const rng = mulberry32(opts.seed ?? 1337);
  const smp = buildSampler(lH, lA, rho);

  let cH = 0, cD = 0, cA = 0, gSum = 0, gSq = 0;
  const scoreCount = new Float64Array(100);
  let total = 0;

  const batch = (nPairs: number) => {
    for (let n = 0; n < nPairs; n++) {
      const u = rng();
      for (const uu of [u, 1 - u]) {          // antithetic pair
        const code = sample(smp, Math.min(uu, 0.999999));
        const i = (code / 10) | 0, j = code % 10;
        if (i > j) cH++; else if (i === j) cD++; else cA++;
        const g = i + j;
        gSum += g; gSq += g * g;
        scoreCount[code]++;
      }
    }
    total += nPairs * 2;
  };

  batch(minPairs);

  // sequential stopping: check CI on max-variance outcome
  const ciHalf = (c: number) => 1.96 * Math.sqrt(Math.max((c / total) * (1 - c / total) / total, 1e-12));
  while (total < maxPairs * 2 && Math.max(ciHalf(cH), ciHalf(cD), ciHalf(cA)) > tol) {
    batch(minPairs);
  }

  const probs = { home: cH / total, draw: cD / total, away: cA / total };
  const ci95 = { home: ciHalf(cH), draw: ciHalf(cD), away: ciHalf(cA) };
  const meanGoals = gSum / total;
  const sdGoals = Math.sqrt(Math.max(gSq / total - meanGoals * meanGoals, 0));

  const scoreFreq = Array.from(scoreCount)
    .map((c, code) => ({ s: `${(code / 10) | 0}:${code % 10}`, i: (code / 10) | 0, j: code % 10, p: c / total }))
    .sort((a, b) => b.p - a.p).slice(0, 10);

  // --- disagreement vs grid (v8.7 bands preserved) ---
  const deltas: Record<Outcome, number> = { home: 0, draw: 0, away: 0 };
  let maxDelta = 0;
  if (opts.gridProbs) {
    for (const k of ['home', 'draw', 'away'] as Outcome[]) {
      deltas[k] = probs[k] - opts.gridProbs[k];
      maxDelta = Math.max(maxDelta, Math.abs(deltas[k]));
    }
  }
  const disagreement: DisagreementLevel =
    maxDelta < 0.015 ? 'none' : maxDelta < 0.03 ? 'low' : maxDelta < 0.06 ? 'medium' : 'high';

  const simPick = (Object.entries(probs) as [Outcome, number][]).sort((a, b) => b[1] - a[1])[0][0];
  let recommendation: SimRecommendation = 'none';
  if (opts.gridPick && disagreement !== 'none') {
    if (simPick !== opts.gridPick && (disagreement === 'high' || disagreement === 'medium'))
      recommendation = 'flip_pick';
    else if (simPick === opts.gridPick && disagreement === 'high')
      recommendation = 'investigate';
    else recommendation = 'shrink_confidence';
  } else if (disagreement !== 'none') recommendation = 'shrink_confidence';

  const confMultiplier = disagreement === 'high' ? 0.93
    : disagreement === 'medium' ? 0.97
    : disagreement === 'low' ? 0.99 : 1.0;

  return {
    n: total, probs, ci95, meanGoals, sdGoals, scoreFreq,
    deltas, maxDelta, disagreement, recommendation, confMultiplier,
    convergedAt: total,
  };
}

// Parlay simulation (v8.7 spec, antithetic-enhanced)
export function simulateParlay(
  legs: { lH: number; lA: number; rho: number; pick: Outcome }[],
  nPairs = 5000, seed = 42,
) {
  const rng = mulberry32(seed);
  const samplers = legs.map((l) => buildSampler(l.lH, l.lA, l.rho));
  const legHits = legs.map(() => 0);
  let allHit = 0;
  const dist = new Float64Array(legs.length + 1);
  const total = nPairs * 2;

  for (let n = 0; n < nPairs; n++) {
    const u = rng();
    for (const uu of [u, 1 - u]) {
      let correct = 0;
      legs.forEach((leg, k) => {
        const code = sample(samplers[k], Math.min(uu, 0.999999));
        const i = (code / 10) | 0, j = code % 10;
        const res: Outcome = i > j ? 'home' : i === j ? 'draw' : 'away';
        if (res === leg.pick) { legHits[k]++; correct++; }
      });
      if (correct === legs.length) allHit++;
      dist[correct]++;
    }
  }
  const cumulative: Record<number, number> = {};
  for (let x = Math.ceil(legs.length / 2); x <= legs.length; x++) {
    let acc = 0;
    for (let k = x; k <= legs.length; k++) acc += dist[k] / total;
    cumulative[x] = acc;
  }
  return {
    allCorrect: allHit / total,
    legProbs: legHits.map((h) => h / total),
    expectedCorrect: legHits.reduce((s, h) => s + h / total, 0),
    distribution: Array.from(dist).map((c) => c / total),
    cumulative,
  };
}

const ORDER: Outcome[] = ['home', 'draw', 'away'];
const KEY: Record<string, string> = {

// theta: rank correlation between half outcomes (0.65 default from
// Bundesliga copula studies 2024-25; league-tunable in Phase 4)
export function htftCopulaJoint(
  lH: number, lA: number, rho: number, theta = 0.65, nSims = 20000, seed = 7,
): HTFTJoint {
  const sm = buildScoreMatrix(lH, lA, rho);
  const ft = outcomeProbs(sm.raw, sm.sum);
  const hts = buildScoreMatrix(lH / 2, lA / 2, rho); // Poisson splitting (half lambdas)

// --- Recalibration: Bayesian posterior update with time decay ---
export function recalibrate(
  params: LearnedParams, predictions: PredictionRecord[], gameType: GameType,
): { params: LearnedParams; message: string } {
  const resolved = predictions
    .filter((p) => p.gameType === gameType && p.result !== undefined)
    .sort((a, b) => (a.resolvedAt ?? 0) - (b.resolvedAt ?? 0));

  if (resolved.length < MIN_RECALIBRATION_RESULTS)
    return { params, message: `Insufficient data: ${resolved.length}/${MIN_RECALIBRATION_RESULTS} resolved` };

  const fresh = defaultParams();
  fresh.totalPredictions = predictions.filter((p) => p.gameType === gameType).length;
  fresh.totalResolved = resolved.length;
  fresh.leagueDrawBoost = { ...params.leagueDrawBoost };
  fresh.leagueUpsetFreq = { ...params.leagueUpsetFreq };

  let hits = 0;
  const outcomeHits: Record<Outcome, number> = { home: 0, draw: 0, away: 0 };
  const outcomeSeen: Record<Outcome, number> = { home: 0, draw: 0, away: 0 };

  resolved.forEach((p, idx) => {
    const age = resolved.length - 1 - idx;
    const w = Math.exp(-DECAY * age);                 // recent results weigh more
    const actual = p.result!;
    const hit = p.pick === actual;
    if (hit) { hits++; fresh.posteriors[actual].alpha += w; }
    else fresh.posteriors[p.pick].beta += w;          // picked outcome failed
    outcomeSeen[actual]++;
    if (hit) outcomeHits[actual]++;
  });

  // derive bounded adjustments from posterior surprise vs baseline
  const drawPost = fresh.posteriors.draw;
  const drawSurprise = drawPost.alpha / (drawPost.alpha + drawPost.beta) - 0.30;
  fresh.drawBoost = Math.min(Math.max(drawSurprise, 0) * 0.25, 0.04);

  const awayPost = fresh.posteriors.away;
  const upsetSurprise = awayPost.alpha / (awayPost.alpha + awayPost.beta) - 0.28;
  fresh.upsetWeight = Math.min(Math.max(upsetSurprise, 0) * 0.2, 0.03);

  const sofPicks = resolved.filter((p) => p.sofTriggered);
  if (sofPicks.length >= 5) {
    const sofHit = sofPicks.filter((p) => p.pick === p.result).length / sofPicks.length;
    fresh.sofThreshold = Math.min(Math.max(0.08 - (sofHit - 0.5) * 0.02, 0.05), 0.12);
  } else fresh.sofThreshold = params.sofThreshold;

  const tight = resolved.filter((p) => p.margin < 0.15);
  if (tight.length >= 5) {
    const tightHit = tight.filter((p) => p.pick === p.result).length / tight.length;
    fresh.rhoAdjust = tightHit < 0.3 ? -0.01 : tightHit > 0.5 ? 0.01 : 0;
  }

  fresh.hitRates = {
    home: outcomeSeen.home ? outcomeHits.home / outcomeSeen.home : 0,
    draw: outcomeSeen.draw ? outcomeHits.draw / outcomeSeen.draw : 0,
    away: outcomeSeen.away ? outcomeHits.away / outcomeSeen.away : 0,
    overall: hits / resolved.length,
  };
  const recent = resolved.slice(-Math.min(50, resolved.length));
  fresh.recentHitRate = recent.filter((p) => p.pick === p.result).length / recent.length;
  const older = resolved.slice(0, Math.max(0, resolved.length - 50));
  const olderRate = older.length
    ? older.filter((p) => p.pick === p.result).length / older.length : fresh.recentHitRate;
  fresh.trend = fresh.recentHitRate > olderRate + 0.03 ? 'improving'
    : fresh.recentHitRate < olderRate - 0.03 ? 'declining' : 'stable';

  return { params: fresh, message: `Recalibrated on ${resolved.length} results (beta-binomial, decay=${DECAY})` };
}

// --- Apply adjustments (v8.7 interface preserved; sim multiplier last) ---
export function applyLearnedAdjustments(
  dcH: number, dcD: number, dcA: number,
  league: string, params: LearnedParams, simConfMultiplier = 1.0,
): [number, number, number] {
  let h = dcH, d = dcD, a = dcA;
  const boost = params.drawBoost + (params.leagueDrawBoost[league] ?? 0);
  if (boost > 0) { d += boost; const f = boost / (h + a); h -= h * f; a -= a * f; }
  const uw = params.upsetWeight;
  if (uw > 0) { a += uw; const f = uw / (h + d); h -= h * f; d -= d * f; }
  const s = h + d + a; h /= s; d /= s; a /= s;
  if (simConfMultiplier !== 1.0) {
    const U = 1 / 3;
    h = U + (h - U) * simConfMultiplier;
    d = U + (d - U) * simConfMultiplier;
    a = U + (a - U) * simConfMultiplier;
  }
  return [h, d, a];
}

// ============================================================
// Fine-grained reactive computation graph. Replaces monolithic
// atlasAnalyse() recompute: changing one odd invalidates ONLY
// downstream nodes (calibration -> matrix -> markets -> portfolio).
// ============================================================
type Subscriber = () => void;
let activeComputation: Computed<unknown> | null = null;

export class Signal<T> {
  private subs = new Set<Subscriber>();
  constructor(private value: T) {}
  get(): T {
    if (activeComputation) this.subs.add(activeComputation.recompute);
    return this.value;
  }
  set(v: T) {
    if (Object.is(v, this.value)) return;
    this.value = v;
    [...this.subs].forEach((fn) => fn());
  }
}

export class Computed<T> {
  private cached: T | undefined;
  private dirty = true;
  private subs = new Set<Subscriber>();
  readonly recompute: Subscriber = () => {
    if (!this.dirty) {
      this.dirty = true;
      [...this.subs].forEach((fn) => fn());
    }
  };
  constructor(private fn: () => T) {}
  get(): T {
    if (activeComputation) this.subs.add(activeComputation.recompute);
    if (this.dirty) {
      const prev = activeComputation;
      activeComputation = this;
      try { this.cached = this.fn(); } finally { activeComputation = prev; }
      this.dirty = false;
    }
    return this.cached as T;
  }
}

export const signal = <T>(v: T) => new Signal(v);
export const computed = <T>(fn: () => T) => new Computed(fn);

const DB_NAME = 'atlas_v9';
const STORE = 'kv';

function open(): Promise<IDBDatabase> {
  return new Promise((res, rej) => {
    const req = indexedDB.open(DB_NAME, 1);
    req.onupgradeneeded = () => req.result.createObjectStore(STORE);
    req.onsuccess = () => res(req.result);
    req.onerror = () => rej(req.error);
  });
}

export async function idbGet<T>(key: string): Promise<T | null> {
  const db = await open();
  return new Promise((res) => {
    const r = db.transaction(STORE).objectStore(STORE).get(key);
    r.onsuccess = () => res((r.result as T) ?? null);
    r.onerror = () => res(null);
  });
}

export async function idbSet(key: string, value: unknown): Promise<void> {
  const db = await open();
  return new Promise((res, rej) => {
    const tx = db.transaction(STORE, 'readwrite');
    tx.objectStore(STORE).put(value, key);
    tx.oncomplete = () => res();
    tx.onerror = () => rej(tx.error);
  });
}

export async function idbClear(): Promise<void> {
  const db = await open();
  return new Promise((res) => {
    const tx = db.transaction(STORE, 'readwrite');
    tx.objectStore(STORE).clear();
    tx.oncomplete = () => res();
  });
}
