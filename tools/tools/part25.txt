'use client';
import { useRef, useState } from 'react';
import { atlasAnalyseV9 } from '@/lib/v9/engines/standard';
import { computeExtendedMarketsV9 } from '@/lib/v9/markets/extended-markets';
import { DC_RHO_STANDARD, EMPTY_USER_ODDS } from '@/lib/v9/atlas-types';
import type { Match, Outcome } from '@/lib/v9/atlas-types';
import { fmtPct } from '@/lib/v9/ui/atlas-colors';

interface Report {
  total: number; analyzed: number;
  pickHits: number; pickProfit: number;
  brier: number;
  flags: number; flagHits: number; flagProfit: number;
  flaglessProfit: number;
}

function settleKey(key: string, i: number, j: number): boolean | null {
  let m = key.match(/^over(\d+)$/); if (m) return i + j > +m[1];
  m = key.match(/^under(\d+)$/); if (m) return i + j < +m[1];
  if (key === 'bttsYes') return i > 0 && j > 0;
  if (key === 'bttsNo') return !(i > 0 && j > 0);
  if (key === 'dc1X') return i >= j;
  if (key === 'dcX2') return j >= i;
  if (key === 'dc12') return i !== j;
  m = key.match(/^exact(\d+)$/); if (m) return i + j === +m[1];
  if (key === 'exact6plus') return i + j >= 6;
  if (key === 'oddGoals') return (i + j) % 2 === 1;
  if (key === 'evenGoals') return (i + j) % 2 === 0;
  m = key.match(/^group(\d)(\d)$/); if (m) return i + j >= +m[1] && i + j <= +m[2];
  if (key === 'group6p') return i + j >= 6;
  m = key.match(/^cs_(\d)_(\d)$/); if (m) return i === +m[1] && j === +m[2];
  return null;
}

export default function StandardLabPanel() {
  const fileRef = useRef<HTMLInputElement>(null);
  const [report, setReport] = useState<Report | null>(null);
  const [busy, setBusy] = useState(false);
  const [fileName, setFileName] = useState('');

  const run = async (text: string) => {
    setBusy(true);
    setReport(null);
    await new Promise((r) => setTimeout(r, 30));

    const lines = text.split(/\r?\n/).filter((l) => l.trim());
    const hs = lines[0].split(',');
    const col = (name: string) => hs.indexOf(name);
    const first = (...names: string[]) => { for (const n of names) { const i = col(n); if (i >= 0) return i; } return -1; };
    const iDiv = first('Div'), iHome = first('HomeTeam'), iAway = first('AwayTeam');
    const iHG = first('FTHG'), iAG = first('FTAG'), iR = first('FTR');
    const iH = first('B365H', 'PSH', 'BbAvH', 'WHH');
    const iD = first('B365D', 'PSD', 'BbAvD', 'WHD');
    const iA = first('B365A', 'PSA', 'BbAvA', 'WHA');
    const iO25 = hs.findIndex((h) => h === 'B365>2.5' || h === 'PS>2.5' || h.endsWith('>2.5'));
    const iU25 = hs.findIndex((h) => h === 'B365<2.5' || h === 'PS<2.5' || h.endsWith('<2.5'));
    const iBTH = hs.findIndex((h) => h === 'B365BTH' || /BTH$/.test(h));
    const iBTA = hs.findIndex((h) => h === 'B365BTA' || /BTA$/.test(h));

    const rep: Report = { total: 0, analyzed: 0, pickHits: 0, pickProfit: 0, brier: 0,
                          flags: 0, flagHits: 0, flagProfit: 0, flaglessProfit: 0 };

    const rows = lines.slice(1);
    for (let b = 0; b < rows.length; b += 25) {           // Rule 10.3 batching
      for (const line of rows.slice(b, b + 25)) {
        const c = line.split(',');
        rep.total++;
        const h = parseFloat(c[iH]), d = parseFloat(c[iD]), a = parseFloat(c[iA]);
        const hg = parseInt(c[iHG], 10), ag = parseInt(c[iAG], 10);
        const ftr = (c[iR] ?? '').trim();
        if (!(h > 1 && d > 1 && a > 1) || !(hg >= 0) || !(ag >= 0) || !ftr) continue;
        try {
          const userOdds: Record<string, number> = { ...EMPTY_USER_ODDS };
          if (iO25 >= 0 && parseFloat(c[iO25]) > 1) userOdds.over25 = parseFloat(c[iO25]);
          if (iU25 >= 0 && parseFloat(c[iU25]) > 1) userOdds.under25 = parseFloat(c[iU25]);
          if (iBTH >= 0 && parseFloat(c[iBTH]) > 1) userOdds.bttsYes = parseFloat(c[iBTH]);
          if (iBTA >= 0 && parseFloat(c[iBTA]) > 1) userOdds.bttsNo = parseFloat(c[iBTA]);
          const match: Match = {
            id: `lab_${rep.total}`, home: c[iHome] ?? '', away: c[iAway] ?? '',
            league: c[iDiv] ?? '', country: '', kickoff: '',
            h: String(h), d: String(d), a: String(a),
            gameType: 'standard', userOdds,
          };
          const an = atlasAnalyseV9(match, 0.10);
          const ext = computeExtendedMarketsV9(
            an.scoreMatrix, an.calibration.lambdaH, an.calibration.lambdaA,
            DC_RHO_STANDARD, userOdds,
          );
          rep.analyzed++;

          const o: Outcome = ftr === 'H' ? 'home' : ftr === 'A' ? 'away' : 'draw';
          const pickOdds = an.pick === 'home' ? h : an.pick === 'away' ? a : d;
          const hit = an.pick === o;
          rep.pickHits += hit ? 1 : 0;
          rep.pickProfit += hit ? pickOdds - 1 : -1;
          for (const k of ['home', 'draw', 'away'] as Outcome[])
            rep.brier += (an.probs[k] - (k === o ? 1 : 0)) ** 2;

          const flag = ext.topEvBets.length ? ext.topEvBets[0] : null;
          if (flag && flag.odds > 1) {
            const s = settleKey(flag.key, hg, ag);
            if (s !== null) {
              rep.flags++;
              rep.flagHits += s ? 1 : 0;
              rep.flagProfit += s ? flag.odds - 1 : -1;
            }
          }
        } catch { /* skip malformed */ }
      }
      await new Promise((r) => setTimeout(r, 0));
    }

    setReport(rep);
    setBusy(false);
  };

  const card = (v: string, l: string, tone = 'text-zinc-100') => (
    <div className="rounded-xl border border-zinc-800 bg-zinc-900/60 p-3 text-center">
      <div className={`text-lg font-bold ${tone}`}>{v}</div>
      <div className="text-[9px] text-zinc-500">{l}</div>
    </div>
  );

  return (
    <div className="space-y-4">
      <div className="rounded-xl border border-blue-500/25 bg-zinc-900/60 p-3">
        <div className="mb-2 flex flex-wrap items-center justify-between gap-2">
          <h3 className="text-xs font-bold text-blue-400">
            STANDARD LAB — out-of-sample validation (football-data.co.uk CSV)
          </h3>
          <button onClick={() => fileRef.current?.click()}
            className="rounded-lg border border-zinc-700 px-2 py-1 text-[10px] text-zinc-300 hover:bg-zinc-800">
            Upload season CSV
          </button>
          <input ref={fileRef} type="file" accept=".csv" className="hidden"
            onChange={async (e) => {
              const f = e.target.files?.[0];
              if (f) { setFileName(f.name); run(await f.text()); }
            }} />
        </div>
        <div className="text-[10px] text-zinc-500">
          {fileName || 'Suggested: E0.csv (England 25/26) from football-data.co.uk — then SP1, D1, I1, F1 for volume.'}
          {' '}Every +EV flag is settled against the real score at flat 1-unit stakes.
        </div>
      </div>

      {busy && <div className="text-xs text-zinc-400">Crunching season… (batched, UI stays live)</div>}

      {report && (
        <>
          <div className="grid grid-cols-3 gap-3 md:grid-cols-6">
            {card(`${report.analyzed}`, 'matches analyzed')}
            {card(fmtPct(report.analyzed ? report.pickHits / report.analyzed : 0), '1X2 pick hit rate', 'text-blue-400')}
            {card((report.analyzed ? report.brier / report.analyzed : 0).toFixed(3), 'Brier (baseline 0.667)')}
            {card(`${report.flags}`, 'extended +EV flags raised', 'text-amber-400')}
            {card(report.flags ? fmtPct(report.flagHits / report.flags) : '—', 'flag hit rate', 'text-amber-400')}
            {card(report.flags ? `${report.flagProfit >= 0 ? '+' : ''}${report.flagProfit.toFixed(1)}u` : '—',
              'flag P/L (flat 1u)', report.flagProfit > 0 ? 'text-emerald-400' : 'text-rose-400')}
          </div>
          <div className="rounded-xl border border-zinc-800 bg-zinc-900/60 p-3 text-[11px] text-zinc-400">
            Baseline P/L if you had blindly bet every 1X2 pick:{' '}
            <b className={report.pickProfit >= 0 ? 'text-emerald-400' : 'text-rose-400'}>
              {report.pickProfit >= 0 ? '+' : ''}{report.pickProfit.toFixed(1)}u
            </b>
            {' '}over {report.analyzed} bets. The flag P/L vs this baseline is the Standard engine's real verdict:
            flags should earn while blind picks lose ~the margin.
          </div>
        </>
      )}
    </div>
  );
}

'use client';
import { useRef, useState } from 'react';
import { useAtlasStore } from '@/lib/v9/store/atlas-store';
import MatchCard from '@/components/atlas/MatchCard';
import QuickViewTables from '@/components/atlas/QuickViewTables';
import SummaryPanel from '@/components/atlas/SummaryPanel';
import SimulationPanel from '@/components/atlas/SimulationPanel';
import LearningPanel from '@/components/atlas/LearningPanel';
import CalibrationPanel from '@/components/atlas/CalibrationPanel';
import APIPicksPanel from '@/components/atlas/APIPicksPanel';
import ScreenerPanel from '@/components/atlas/ScreenerPanel';
import JackpotLabPanel from '@/components/atlas/JackpotLabPanel';
import StandardLabPanel from '@/components/atlas/StandardLabPanel';
import ScrapeButton from '@/components/atlas/ScrapeButton';

const TABS = ['Matches', 'Screener', 'Jackpot Lab', 'Standard Lab', 'Quick View', 'Simulations', 'Learning', 'Backtest', 'API Picks'] as const;

export default function AtlasPage() {
  const [tab, setTab] = useState<(typeof TABS)[number]>('Matches');
  const matches = useAtlasStore((s) => s.matches);
  const phi = useAtlasStore((s) => s.phi);
  const setPhi = useAtlasStore((s) => s.setPhi);
  const addMatch = useAtlasStore((s) => s.addMatch);
  const exportSession = useAtlasStore((s) => s.exportSession);
  const importSession = useAtlasStore((s) => s.importSession);
  const fileRef = useRef<HTMLInputElement>(null);

  const download = () => {
    const blob = new Blob([exportSession()], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = `atlas_v9_session_${new Date().toISOString().slice(0, 10)}.json`;
    a.click();
  };

  const clearAll = () => {
    if (window.confirm(`Delete ALL ${matches.length} matches? Results and learning data are kept. This cannot be undone.`)) {
      useAtlasStore.setState({ matches: [] });
    }
  };

  return (
    <div className="min-h-screen bg-zinc-950 p-4 md:p-6 text-zinc-100">
      <header className="mb-5 flex flex-wrap items-center justify-between gap-3">
        <div>
          <h1 className="text-xl font-bold tracking-tight">
            ATLAS <span className="text-blue-400">v9.9</span>
            <span className="ml-2 text-xs font-normal text-zinc-500">
              Enhanced · hybrid Bayesian calibration · copula HT/FT · adaptive MC · beta-binomial learning · fixture screener · jackpot lab · standard lab
            </span>
          </h1>
        </div>
        <div className="flex flex-wrap items-center gap-2">
          <label className="flex items-center gap-2 rounded-lg border border-zinc-800 bg-zinc-900 px-3 py-1.5 text-xs">
            φ threshold <b className="tabular-nums text-blue-400">{phi.toFixed(2)}</b>
            <input type="range" min={0} max={0.3} step={0.01} value={phi}
              onChange={(e) => setPhi(parseFloat(e.target.value))} className="w-24 accent-blue-500" />
          </label>
          <button onClick={addMatch} className="rounded-lg bg-blue-600 px-3 py-1.5 text-xs font-semibold hover:bg-blue-500">+ Add Match</button>
          <button onClick={download} className="rounded-lg border border-zinc-700 px-3 py-1.5 text-xs text-zinc-300 hover:bg-zinc-800">Export</button>
          <button onClick={() => fileRef.current?.click()} className="rounded-lg border border-zinc-700 px-3 py-1.5 text-xs text-zinc-300 hover:bg-zinc-800">Import</button>
          <input ref={fileRef} type="file" accept=".json" className="hidden" onChange={async (e) => {
            const f = e.target.files?.[0];
            if (f) importSession(await f.text());
          }} />
        </div>
      </header>

      <nav className="mb-4 flex flex-wrap items-center gap-1.5">
        {TABS.map((t) => (
          <button key={t} onClick={() => setTab(t)}
            className={`rounded-lg px-3 py-1.5 text-xs font-medium ${
              tab === t ? 'bg-zinc-100 text-zinc-950' : 'bg-zinc-900 text-zinc-400 hover:bg-zinc-800'}`}>
            {t}
          </button>
        ))}
        <div className="ml-auto"><ScrapeButton /></div>
      </nav>

      {tab === 'Matches' && (
        <div className="space-y-4">
          <div className="flex justify-end">
            {matches.length > 0 && (
              <button onClick={clearAll}
                className="rounded-lg border border-rose-500/30 px-3 py-1.5 text-[11px] text-rose-400 hover:bg-rose-500/10">
                🗑 Clear All Matches
              </button>
            )}
          </div>
          <SummaryPanel />
          {matches.length === 0 && (
            <div className="rounded-xl border border-dashed border-zinc-800 p-10 text-center text-sm text-zinc-600">
              No matches yet. Click “+ Add Match”, use the Screener to import a fixture list, or import API picks.
            </div>
          )}
          {matches.map((m) => <MatchCard key={m.id} match={m} />)}
        </div>
      )}
      {tab === 'Screener' && <ScreenerPanel />}
      {tab === 'Jackpot Lab' && <JackpotLabPanel />}
      {tab === 'Standard Lab' && <StandardLabPanel />}
      {tab === 'Quick View' && <QuickViewTables />}
      {tab === 'Simulations' && <SimulationPanel />}
      {tab === 'Learning' && <LearningPanel />}
      {tab === 'Backtest' && <CalibrationPanel />}
      {tab === 'API Picks' && <APIPicksPanel />}

      <footer className="mt-8 text-center text-[10px] text-zinc-700">
        ATLAS v9.9 · 100% client-side computation · Standard ρ=-0.13 / Jackpot ρ=-0.08 · MC is a cross-check, never the primary model
      </footer>
    </div>
  );
}
