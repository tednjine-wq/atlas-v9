// ============================================================
// Real-World Context engine — tier 1 (free, offline, rule-based).
// Derives narrative sentences from the model itself, exactly like
// v8.4's "Heavy mismatch — strong favorite expected to dominate".
// Tier 2 (LLM) lives in /api/ai-context and feeds match.prior.
// ============================================================
import type { AtlasAnalysis } from '../atlas-types';

export interface ContextLine { text: string; tone: 'info' | 'warn' | 'good'; }

export function buildRuleNarrative(a: AtlasAnalysis): ContextLine[] {
  const lines: ContextLine[] = [];
  const { home, draw, away } = a.dcProbs;
  const gap = Math.abs(home - away);
  const maxP = Math.max(home, draw, away);

  // matchup shape
  if (maxP > 0.60 && gap > 0.30)
    lines.push({ text: 'Heavy mismatch — strong favorite expected to dominate', tone: 'info' });
  else if (maxP > 0.52)
    lines.push({ text: 'Clear favorite — but the opposition is capable of an upset', tone: 'info' });
  else if (gap < 0.08)
    lines.push({ text: 'Evenly matched — coin-flip territory, the draw is very live', tone: 'warn' });
  else
    lines.push({ text: 'Competitive fixture — narrow edge to the favorite', tone: 'info' });

  // scoring tempo from calibrated lambdas
  const xg = a.calibration.lambdaH + a.calibration.lambdaA;
  if (xg > 3.2)
    lines.push({ text: `Open game expected (xG total ${xg.toFixed(2)}) — over profile`, tone: 'good' });
  else if (xg < 2.2)
    lines.push({ text: `Tight, low-scoring affair expected (xG total ${xg.toFixed(2)}) — under profile`, tone: 'info' });
  else
    lines.push({ text: `Moderate scoring tempo expected (xG total ${xg.toFixed(2)})`, tone: 'info' });

  // draw proneness
  if (draw > 0.29)
    lines.push({ text: 'Draw-prone profile — a low-scoring stalemate is a realistic outcome', tone: 'warn' });
  if (a.sofTriggered)
    lines.push({ text: 'SOF v4 active — statistical profile flags the draw', tone: 'warn' });

  // BTTS profile from the DC-corrected matrix (audit-safe, not independence)
  let btts = 0;
  for (const e of a.scoreMatrix) if (e.i > 0 && e.j > 0) btts += e.p;
  if (btts > 0.60)
    lines.push({ text: `Both teams likely to score (${(btts * 100).toFixed(0)}%)`, tone: 'good' });
  else if (btts < 0.42)
    lines.push({ text: `At least one blanking likely (BTTS only ${(btts * 100).toFixed(0)}%)`, tone: 'info' });

  // bookmaker margin commentary
  if (a.marginWarning.msg)
    lines.push({ text: a.marginWarning.msg, tone: 'warn' });

  return lines;
}

// ============================================================
// AI Real-World Context (tier 2, OPTIONAL).
// OpenAI-compatible proxy (Groq free tier by default).
// Returns narrative + a bounded strength prior that feeds the
// v9 hybrid Bayesian calibration. RULE 7 PRESERVED: prediction
// math stays client-side; this only adds context + weak prior.
// Env: AI_API_KEY (required) · AI_BASE_URL · AI_MODEL (optional)
// ============================================================
import { NextResponse } from 'next/server';

export const runtime = 'nodejs';

const clamp = (x: unknown, lo: number, hi: number) =>
  Math.min(hi, Math.max(lo, typeof x === 'number' && isFinite(x) ? x : lo));

export async function POST(req: Request) {
  try {
    const { home, away, league, kickoff } = await req.json();
    const key = process.env.AI_API_KEY;
    const base = process.env.AI_BASE_URL ?? 'https://api.groq.com/openai/v1';
    const model = process.env.AI_MODEL ?? 'llama-3.1-8b-instant';
    if (!key)
      return NextResponse.json({ ai: null, reason: 'AI_API_KEY not set — rule-based context used' });

    const prompt = `You are a senior football analyst. Match: ${home || 'Home'} vs ${away || 'Away'}${league ? ` (${league})` : ''}${kickoff ? `, kickoff ${kickoff}` : ''}.
Return ONLY a valid JSON object (no markdown) with exactly these keys:
{"narrative": "2-3 sentences: recent form, motivation, known injuries/suspensions, tactical style",
 "pHome": 0.0-1.0, "pDraw": 0.0-1.0, "pAway": 0.0-1.0,
 "confidence": 0-8}
The three probabilities must sum to 1. confidence = how sure you are (0-2 low, 3-5 medium, 6-8 high).
If you do not reliably know these teams, set confidence to 0 and narrative to "Insufficient recent information on these teams."`;

    const res = await fetch(`${base}/chat/completions`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${key}` },
      body: JSON.stringify({
        model, temperature: 0.3, max_tokens: 400,
        messages: [{ role: 'user', content: prompt }],
      }),
    });
    if (!res.ok) return NextResponse.json({ ai: null, reason: `AI HTTP ${res.status}` });
    const data = await res.json();
    const text: string = data?.choices?.[0]?.message?.content ?? '';
    const m = text.match(/\{[\s\S]*\}/);
    if (!m) return NextResponse.json({ ai: null, reason: 'AI returned non-JSON' });
    const p = JSON.parse(m[0]);
    let pH = clamp(p.pHome, 0.02, 0.96);
    let pD = clamp(p.pDraw, 0.02, 0.96);
    let pA = clamp(p.pAway, 0.02, 0.96);
    const s = pH + pD + pA; pH /= s; pD /= s; pA /= s;
    return NextResponse.json({
      ai: {
        narrative: String(p.narrative ?? ''),
        pHome: pH, pDraw: pD, pAway: pA,
        confidence: clamp(p.confidence, 0, 8),
      },
    });
  } catch (e) {
    return NextResponse.json({ ai: null, reason: e instanceof Error ? e.message : 'AI error' });
  }
}
