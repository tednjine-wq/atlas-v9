export interface LearnedParams {
  version: number;
  totalPredictions: number;
  totalResolved: number;
  posteriors: Record<Outcome, BetaParams>;
  drawBoost: number;
  sofThreshold: number;
  rhoAdjust: number;
  upsetWeight: number;
  hitRates: Record<Outcome, number> & { overall: number };

  leagueUpsetFreq: Record<string, number>;

const DECAY = 0.06;

export const defaultParams = (): LearnedParams => ({
  version: PARAMS_VERSION, totalPredictions: 0, totalResolved: 0,
  posteriors: {
    home: { alpha: 1, beta: 1 }, draw: { alpha: 1, beta: 1 }, away: { alpha: 1, beta: 1 },
  },
  drawBoost: 0, sofThreshold: 0.08, rhoAdjust: 0, upsetWeight: 0,
  hitRates: { home: 0, draw: 0, away: 0, overall: 0 },
  recentHitRate: 0, trend: 'stable', leagueDrawBoost: {}, leagueUpsetFreq: {},
});

export function recalibrate(
  params: LearnedParams, predictions: PredictionRecord[], gameType: GameType,
): { params: LearnedParams; message: string } {
  const resolved = predictions
    .filter((p) => p.gameType === gameType && p.result !== undefined)
    .sort((a, b) => (a.resolvedAt ?? 0) - (b.resolvedAt ?? 0));

  if (resolved.length < MIN_RECALIBRATION_RESULTS)
    return { params, message: `Insufficient data: ${resolved.length}/${MIN_RECALIBRATION_RESULTS} resolved` };

  const fresh = defaultParams();
  fresh.totalPredictions = predictions.filter((p) => p.gameType === gameType).length;
  fresh.totalResolved = resolved.length;
  fresh.leagueDrawBoost = { ...params.leagueDrawBoost };
  fresh.leagueUpsetFreq = { ...params.leagueUpsetFreq };

  let hits = 0;
  const outcomeHits: Record<Outcome, number> = { home: 0, draw: 0, away: 0 };
  const outcomeSeen: Record<Outcome, number> = { home: 0, draw: 0, away: 0 };

  resolved.forEach((p, idx) => {
    const age = resolved.length - 1 - idx;
    const w = Math.exp(-DECAY * age);
    const actual = p.result!;
    const hit = p.pick === actual;
    if (hit) { hits++; fresh.posteriors[actual].alpha += w; }
    else fresh.posteriors[p.pick].beta += w;
    outcomeSeen[actual]++;
    if (hit) outcomeHits[actual]++;
  });

  const drawPost = fresh.posteriors.draw;
  const drawSurprise = drawPost.alpha / (drawPost.alpha + drawPost.beta) - 0.30;
  fresh.drawBoost = Math.min(Math.max(drawSurprise, 0) * 0.25, 0.04);

  const awayPost = fresh.posteriors.away;
  const upsetSurprise = awayPost.alpha / (awayPost.alpha + awayPost.beta) - 0.28;
  fresh.upsetWeight = Math.min(Math.max(upsetSurprise, 0) * 0.2, 0.03);

  const sofPicks = resolved.filter((p) => p.sofTriggered);
  if (sofPicks.length >= 5) {
    const sofHit = sofPicks.filter((p) => p.pick === p.result).length / sofPicks.length;
    fresh.sofThreshold = Math.min(Math.max(0.08 - (sofHit - 0.5) * 0.02, 0.05), 0.12);
  } else fresh.sofThreshold = params.sofThreshold;

  const tight = resolved.filter((p) => p.margin < 0.15);
  if (tight.length >= 5) {
    const tightHit = tight.filter((p) => p.pick === p.result).length / tight.length;
    fresh.rhoAdjust = tightHit < 0.3 ? -0.01 : tightHit > 0.5 ? 0.01 : 0;
  }

  fresh.hitRates = {
    home: outcomeSeen.home ? outcomeHits.home / outcomeSeen.home : 0,
    draw: outcomeSeen.draw ? outcomeHits.draw / outcomeSeen.draw : 0,
    away: outcomeSeen.away ? outcomeHits.away / outcomeSeen.away : 0,
    overall: hits / resolved.length,
  };
  const recent = resolved.slice(-Math.min(50, resolved.length));
  fresh.recentHitRate = recent.filter((p) => p.pick === p.result).length / recent.length;
  const older = resolved.slice(0, Math.max(0, resolved.length - 50));
  const olderRate = older.length
    ? older.filter((p) => p.pick === p.result).length / older.length : fresh.recentHitRate;
  fresh.trend = fresh.recentHitRate > olderRate + 0.03 ? 'improving'
    : fresh.recentHitRate < olderRate - 0.03 ? 'declining' : 'stable';

  return { params: fresh, message: `Recalibrated on ${resolved.length} results (beta-binomial, decay=${DECAY})` };
}

export function applyLearnedAdjustments(
  dcH: number, dcD: number, dcA: number,
  league: string, params: LearnedParams, simConfMultiplier = 1.0,
): [number, number, number] {
  let h = dcH, d = dcD, a = dcA;
  const boost = params.drawBoost + (params.leagueDrawBoost[league] ?? 0);
  if (boost > 0) { d += boost; const f = boost / (h + a); h -= h * f; a -= a * f; }
  const uw = params.upsetWeight;
  if (uw > 0) { a += uw; const f = uw / (h + d); h -= h * f; d -= d * f; }
  const s = h + d + a; h /= s; d /= s; a /= s;
  if (simConfMultiplier !== 1.0) {
    const U = 1 / 3;
    h = U + (h - U) * simConfMultiplier;
    d = U + (d - U) * simConfMultiplier;
    a = U + (a - U) * simConfMultiplier;
  }
  return [h, d, a];
}

// STANDARD ENGINE v9.1 — EV-maximizing, rho = -0.13 (RULE 1)
// v9.1: HONEST confidence (no artificial SOF boosts),

import { DC_RHO_STANDARD } from '../atlas-types';
import type {
  Match, Outcome, AtlasAnalysis, MarginWarning, PortfolioOutcome, DecorrelationResult, Verdict,
} from '../atlas-types';
import { calibrateLambda, stripMargin } from '../math/calibration';
import { buildScoreMatrix, outcomeProbs } from '../math/dixon-coles';
import { adaptiveSimulate } from '../math/adaptive-mc';
import { htftCopulaJoint } from '../math/copula';

function marginWarning(minOdds: number): MarginWarning {
  if (minOdds < 1.25) return { level: 'severe', msg: 'Extreme favorite — bookmaker margin asymmetry severe. Avoid.' };
  if (minOdds < 1.45) return { level: 'high', msg: 'Heavy favorite — elevated bookmaker edge on short side.' };
  if (minOdds < 1.65) return { level: 'moderate', msg: 'Moderate favorite — slightly elevated margin.' };
  return { level: 'none', msg: '' };
}

export function atlasAnalyseV9(
  match: Match, phi = 0.10, learnedParams: LearnedParams = defaultParams(),
): AtlasAnalysis {
  const h = parseFloat(match.h), d = parseFloat(match.d), a = parseFloat(match.a);
  if (!(h > 1 && d > 1 && a > 1)) throw new Error('Invalid 1X2 odds');

  const { probs: bookProbs, margin } = stripMargin(h, d, a);
  const cal = calibrateLambda(h, d, a, DC_RHO_STANDARD, match.prior);
  const rho = DC_RHO_STANDARD + learnedParams.rhoAdjust;

  const sm = buildScoreMatrix(cal.lambdaH, cal.lambdaA, rho);
  const dc = outcomeProbs(sm.raw, sm.sum);

  // v9.1 tight-game flag
  const tightGame = Math.abs(dc.home - dc.away) < 0.08;

  let pick: Outcome = (Object.entries(dc) as [Outcome, number][]).sort((x, y) => y[1] - x[1])[0][0];
  let confidence = dc[pick]; // v9.1: honest — no artificial boosts

  // SOF v4 (pick override kept; confidence stays honest)
  const sofT = learnedParams.sofThreshold;
  const sofPrimary = Math.abs(dc.home - dc.away) < sofT;
  const sofSecondary = Math.abs(dc.home - dc.draw) < 0.03 || Math.abs(dc.draw - dc.away) < 0.03;
  const drawShortest = d < h && d < a;
  const sofTriggered = sofPrimary || sofSecondary || drawShortest;
  if (sofTriggered) {
    pick = 'draw';
    confidence = dc.draw;
  }

  const edges: Record<Outcome, number> = {
    home: dc.home - bookProbs[0], draw: dc.draw - bookProbs[1], away: dc.away - bookProbs[2],
  };
  const oddsMap: Record<Outcome, number> = { home: h, draw: d, away: a };
  const evs: Record<Outcome, number> = {
    home: dc.home * h - 1, draw: dc.draw * d - 1, away: dc.away * a - 1,
  };
  const spots = (Object.keys(edges) as Outcome[]).filter((o) => edges[o] > 0 && evs[o] > 0);
  const spotted = spots.sort((x, y) => edges[y] - edges[x])[0] ?? null;
  const maxEdge = Math.max(...Object.values(edges), 0);
  const decorrelation: DecorrelationResult = {
    score: Math.min(100, maxEdge * 500), edges, evs, spotted,
    hasRealSpot: !!spotted, maxEdge,
  };
  let decorrOverride = false;
  if (spotted && spotted !== pick && decorrelation.score >= 15 && edges[spotted] > 0.04 && evs[spotted] > 0.03) {
    pick = spotted; decorrOverride = true;
    confidence = dc[pick]; // honest
  }
  confidence = Math.min(confidence, 0.78); // v8.7 cap preserved

  const sim = adaptiveSimulate(cal.lambdaH, cal.lambdaA, DC_RHO_STANDARD, {
    gridProbs: dc, gridPick: pick,
  });

  const [fh, fd, fa] = applyLearnedAdjustments(
    dc.home, dc.draw, dc.away, match.league, learnedParams, sim.confMultiplier,
  );
  const probs: Record<Outcome, number> = { home: fh, draw: fd, away: fa };

  const portfolio: PortfolioOutcome[] = (['home', 'draw', 'away'] as Outcome[]).map((o) => {
    const ev = probs[o] * oddsMap[o] - 1;
    const variance = (1 - probs[o]) * probs[o] * oddsMap[o] ** 2;
    const kelly = ev > 0 ? Math.min(ev / (oddsMap[o] - 1), 0.25) : 0;
    return { outcome: o, stake: kelly, ev, sharpe: variance > 0 ? ev / Math.sqrt(variance) : 0,
             kelly, isMaxEp: false, varPerUnit: variance };
  });
  const best = portfolio.filter((p) => p.ev > 0).sort((x, y) => y.ev - x.ev)[0];
  if (best) best.isMaxEp = true;

  const htft = htftCopulaJoint(cal.lambdaH, cal.lambdaA, DC_RHO_STANDARD);
  const thresholdPass = Math.abs(probs[pick] - 0.5) >= phi;

  // v9.1 verdict: BET only with threshold pass AND a positive-EV spot
  const bestEv = Math.max(...portfolio.map((p) => p.ev));
  const verdict: Verdict = thresholdPass && bestEv > 0 ? 'BET' : 'NO-BET';

  return {
    pick, confidence, probs, dcProbs: dc,
    bookProbs: { home: bookProbs[0], draw: bookProbs[1], away: bookProbs[2] },
    margin, marginWarning: marginWarning(Math.min(h, d, a)),
    calibration: cal, scoreMatrix: sm.matrix, decorrelation, portfolio,
    sofTriggered, decorrOverride, thresholdPass, verdict, tightGame,
    sim, htft, engine: 'standard',
  };
}

// JACKPOT ENGINE v9.1 — parlay-max, rho = -0.08 (RULE 1)
// RULE 6 preserved: no decorrelation override, no learning.
// v9.1: verdict + tightGame for type parity.

import { DC_RHO_JACKPOT, DC_RHO_STANDARD } from '../atlas-types';
import type { Match, Outcome, AtlasJackpotAnalysis, DecorrelationResult, Verdict } from '../atlas-types';
import { calibrateLambda, stripMargin } from '../math/calibration';

export function atlasAnalyseJackpotV9(match: Match, phi = 0.10): AtlasJackpotAnalysis {
  const h = parseFloat(match.h), d = parseFloat(match.d), a = parseFloat(match.a);
  if (!(h > 1 && d > 1 && a > 1)) throw new Error('Invalid 1X2 odds');
  const { probs: bookProbs, margin } = stripMargin(h, d, a);
  const cal = calibrateLambda(h, d, a, DC_RHO_JACKPOT, match.prior);
  const { matrix, raw, sum } = buildScoreMatrix(cal.lambdaH, cal.lambdaA, DC_RHO_JACKPOT);
  const dc = outcomeProbs(raw, sum);

  const std = buildScoreMatrix(cal.lambdaH, cal.lambdaA, DC_RHO_STANDARD);
  const dcStd = outcomeProbs(std.raw, std.sum);
  const homeBias = dcStd.home - dc.home;
  const tightGame = Math.abs(dc.home - dc.away) < 0.08;

  const sorted = (Object.entries(dc) as [Outcome, number][]).sort((x, y) => y[1] - x[1]);
  let pick: Outcome = sorted[0][0];
  const secondaryPick: Outcome = sorted[1][0];
  const sofTriggered = Math.abs(dc.home - dc.away) < 0.06 || (d < h && d < a);
  if (sofTriggered) pick = 'draw';

  const edges: Record<Outcome, number> = {
    home: dc.home - bookProbs[0], draw: dc.draw - bookProbs[1], away: dc.away - bookProbs[2],
  };
  const evs: Record<Outcome, number> = {
    home: dc.home * h - 1, draw: dc.draw * d - 1, away: dc.away * a - 1,
  };
  const spots = (Object.keys(edges) as Outcome[])
    .filter((o) => edges[o] > 0 && evs[o] > 0)
    .sort((x, y) => edges[y] - edges[x]);
  const maxEdge = Math.max(...Object.values(edges), 0);
  const skippedDecorrOverride = spots[0] && spots[0] !== pick && Math.min(100, maxEdge * 500) >= 15
    ? { wouldFlipTo: spots[0], evLost: evs[spots[0]] } : null;

  const sim = adaptiveSimulate(cal.lambdaH, cal.lambdaA, DC_RHO_JACKPOT, { gridProbs: dc, gridPick: pick });
  const confidence = Math.min(dc[pick], 0.72);

  const decorrelation: DecorrelationResult = {
    score: Math.min(100, maxEdge * 500),
    edges, evs, spotted: spots[0] ?? null, hasRealSpot: spots.length > 0, maxEdge,
  };

  const thresholdPass = Math.abs(dc[pick] - 0.5) >= phi;
  const verdict: Verdict = thresholdPass ? 'BET' : 'NO-BET'; // parlay-leg suitability

  return {
    pick, confidence,
    probs: dc, dcProbs: dc,
    bookProbs: { home: bookProbs[0], draw: bookProbs[1], away: bookProbs[2] },
    margin, marginWarning: { level: 'none', msg: '' },
    calibration: cal, scoreMatrix: matrix, decorrelation, portfolio: [],
    sofTriggered, decorrOverride: false,
    thresholdPass, verdict, tightGame,
    sim, engine: 'jackpot',
    parlayScore: Math.log(Math.max(dc[pick], 1e-9)),
    upsetVulnerable: Math.abs(dc.home - dc.away) < 0.05,
    secondaryPick, secondaryProb: dc[secondaryPick],
    homeBias, skippedDecorrOverride,
  };
}
