// ============================================================
// Fixture parser v9.19 — glue-rule fixes:
//  • an odds line is appended to an odds buffer ONLY when it
//    carries NO date (vertical duplicate cell); a date-carrying
//    line is always a NEW record (horizontal single-line format)
//  • the header-junk date flush requires buf.length > 6 so
//    vertical fragments ("1 1587" + date) stay together
// Keeps all v9.13–v9.18 behavior.
// ============================================================
export interface FixtureRow {
  home: string; away: string; league: string; country: string; kickoff: string;
  h: string; d: string; a: string;
  userOdds?: Record<string, number>;
}

const GARBAGE = [
  /home\s+draw\s+away/i, /3 way/i, /double chance/i, /over\/under/i, /both teams/i,
  /^id\b/i, /team 1/i, /team 2/i, /\bdate\b/i, /sportpesa/i, /paybill/i,
  /legal/i, /licence/i, /1 or x/i, /x or 2/i, /\byes\s+no\b/i, /quick pick/i,
  /only 18\+/i, /gambling may/i, /licensed and regulated/i, /http/i, /p\.o\.box/i,
];
const isGarbage = (s: string) => GARBAGE.some((r) => r.test(s));
const clean = (s: string) => s.replace(/\|/g, ' ').replace(/\s+/g, ' ').trim();

const COUNTRIES = new Set([
  'Germany', 'Russia', 'Sweden', 'Norway', 'France', 'Belgium', 'Netherlands', 'Austria',
  'Spain', 'England', 'Scotland', 'Wales', 'Denmark', 'Finland', 'Iceland', 'Turkey',
  'Serbia', 'Poland', 'Portugal', 'Czechia', 'Kazakhstan', 'USA', 'Brazil', 'Argentina',
  'Mozambique', 'Chile', 'Japan', 'Australia', 'Croatia', 'Greece', 'Switzerland',
  'Romania', 'Hungary', 'Ukraine', 'Ireland', 'Estonia', 'Latvia', 'Lithuania', 'Slovenia',
]);

export function parseFixtureList(text: string): FixtureRow[] {
  const rows: FixtureRow[] = [];
  let league = '';
  let country = '';
  let date = '';

  const setHeader = (raw: string) => {
    const h = clean(raw);
    if (!h || h.length > 70 || isGarbage(h) || /\d\.\d+/.test(h)) return;
    const m = h.match(/^(.+?)\s*-\s*(.+)$/);
    if (m && m[1].length <= 26 && m[2].length >= 3) {
      country = m[1].trim();
      league = m[2].trim();
    } else {
      league = h;
    }
  };

  const splitTeams = (blob: string): [string, string] => {
    const w = blob.split(' ');
    const n = Math.ceil(w.length / 2);
    let home = w.slice(0, n).join(' ');
    let away = w.slice(n).join(' ');
    const yr = away.match(/^(19|20)\d{2}/);
    if (yr && away.split(' ').length > 1) {
      home += ' ' + yr[0];
      away = away.replace(/^(19|20)\d{2}\s*/, '');
    }
    return [home, away || w[w.length - 1]];
  };

  // ---------- v9.16–19: rebuild vertical cells into logical lines ----------
  const rawLines = text.split(/\n/).map(clean).filter(Boolean);
  const lines: string[] = [];
  let buf: string[] = [];
  const hasOdds = (s: string) => /\d\.\d+/.test(s);
  const hasDate = (s: string) => /\d{2}-\d{2}-\d{4}/.test(s);
  const isInt = (s: string) => /^\d+$/.test(s);
  const flush = () => { if (buf.length) { lines.push(buf.join(' ')); buf = []; } };

  for (const L of rawLines) {
    if (!buf.length) { buf = [L]; continue; }
    const bt = buf.join(' ');
    if (hasOdds(bt)) {
      // v9.19: append ONLY date-less odds cells; dated line = new record
      if (hasOdds(L) && !hasDate(L)) { if (buf[buf.length - 1] !== L) buf.push(L); }
      else { flush(); buf = [L]; }
    } else if (hasDate(bt) || /\d/.test(bt)) {
      const newRec = (/^\d{1,2}$/.test(L) && buf.length > 6) ||
                     (hasDate(L) && !hasDate(bt) && buf.length > 6);
      if (newRec) { flush(); buf = [L]; }
      else if (buf[buf.length - 1] !== L) buf.push(L);
    } else {
      if (isInt(L) || hasDate(L) || hasOdds(L)) { flush(); buf = [L]; }
      else if (buf[buf.length - 1] !== L) buf.push(L);
    }
  }
  flush();

  // ---------- main parse over normalized lines ----------
  for (const flat of lines) {
    if (isGarbage(flat)) continue;
    if (/^\d{2}[\/.-]\d{2}[\/.-]\d{2,4}$/.test(flat)) { date = flat; continue; }

    const nums = (flat.match(/\d+\.\d+/g) || []).map(parseFloat);

    if (flat.includes('|')) {
      const cells = flat.split('|').map((c) => c.trim()).filter((c) => c.length);
      if (cells.length <= 2 && !nums.length) { setHeader(cells[0] ?? ''); continue; }
      const teamCells: string[] = [];
      let kick = '';
      for (const c of cells) {
        if (/^\d{2}-\d{2}-\d{4}/.test(c)) {
          const t = c.match(/\d{1,2}:\d{2}/);
          kick = `${c.slice(0, 10)}${t ? ' ' + t[0] : ''}`;
        } else if (/^\d+$/.test(c)) continue;
        else if (!/\d\.\d+/.test(c)) teamCells.push(c);
      }
      if (nums.length >= 3 && teamCells.length >= 2) {
        let home = teamCells[0];
        let away = teamCells[1];
        let ctry = teamCells[2] ?? '';
        if (teamCells.length === 2 &&
            home.split(' ').length >= 3 && away.split(' ').length <= 2 && !/\d/.test(away)) {
          [home, away] = splitTeams(home);
          ctry = teamCells[1];
        }
        const u: Record<string, number> = {};
        if (nums.length >= 10) {
          u.dc1X = nums[3]; u.dcX2 = nums[4]; u.dc12 = nums[5];
          u.over25 = nums[6]; u.under25 = nums[7];
          u.bttsYes = nums[8]; u.bttsNo = nums[9];
        }
        rows.push({
          home, away, league, country: ctry || country, kickoff: kick || date,
          h: String(nums[0]), d: String(nums[1]), a: String(nums[2]), userOdds: u,
        });
        continue;
      }
      if (!nums.length) setHeader(flat);
      continue;
    }

    const sp = flat.match(
      /^(?:(\d{1,5})\s+)?(?:(\d{1,5})\s+)?(\d{2}-\d{2}-\d{4})\s*-?\s*(\d{1,2}:\d{2})\s+(.+)$/,
    );
    if (sp) {
      const om = sp[5].match(/^(.*?)\s+((?:\d+\.\d+\s+){2,}\d+\.\d+)$/);
      if (om) {
        const od = om[2].split(/\s+/).map(parseFloat);
        if (od.length >= 3) {
          const words = om[1].trim().split(' ');
          let ctry = '';
          if (words.length >= 4 && COUNTRIES.has(words[words.length - 1])) {
            ctry = words.pop() ?? '';
            if (ctry === 'South' && words.length && /^(Africa|Korea)$/i.test(words[words.length - 1]))
              ctry = 'South ' + words.pop();
          }
          const [home, away] = splitTeams(words.join(' '));
          const u: Record<string, number> = {};
          if (od.length >= 9) {
            u.dc1X = od[3]; u.dcX2 = od[4]; u.dc12 = od[5];
            u.over25 = od[6]; u.under25 = od[7]; u.bttsYes = od[8];
            if (od.length >= 10) u.bttsNo = od[9];
          }
          rows.push({
            home, away, league, country: ctry || country, kickoff: `${sp[3]} ${sp[4]}`,
            h: String(od[0]), d: String(od[1]), a: String(od[2]), userOdds: u,
          });
          continue;
        }
      }
    }

    const m = flat.match(
      /^(?:(\d{1,2}:\d{2})\s+)?(.+?)\s+(?:vs\.?|v)\s+(.+?)\s+(\d+\.\d+)\s+(\d+\.\d+)\s+(\d+\.\d+)$/i,
    );
    if (m) {
      rows.push({
        home: m[2].trim(), away: m[3].trim(), league, country,
        kickoff: `${date} ${m[1] ?? ''}`.trim(),
        h: m[4], d: m[5], a: m[6],
      });
      continue;
    }

    if (!nums.length && flat.length < 70) setHeader(flat);
  }
  return rows;
}
